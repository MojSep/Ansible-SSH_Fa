۱️⃣ معماری Control Node — پایه امنیت
اصل: جداسازی و Least Privilege
❌ آنتی‌پترن: اجرای Ansible از laptop شخصی
bash

# روی laptop شخصی مهندس
$ ansible-playbook -i prod site.yml -u root
# مشکلات:
# - کلید شخصی = دسترسی کامل به همه prod
# - لاگ محلی = بدون audit

تنظیمات سخت‌سازی Control Node
bash

# 1. user اختصاصی
sudo useradd -m -s /bin/bash -G wheel ansible
sudo passwd -l ansible           # قفل رمز (فقط SSH key)
sudo usermod -aG docker ansible  # اگر لازم

# 2. sudoers محدود
sudo tee /etc/sudoers.d/ansible <<'EOF'
Defaults:ansible !requiretty
Defaults:ansible !visiblepw
Defaults:ansible env_reset
Defaults:ansible secure_path="/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin"
Defaults:ansible timestamp_timeout=0
ansible ALL=(root) NOPASSWD: /usr/bin/dnf, /usr/bin/apt, /usr/bin/systemctl, /usr/bin/dpkg, /usr/bin/rpm
EOF
sudo chmod 440 /etc/sudoers.d/ansible
sudo visudo -cf /etc/sudoers.d/ansible

# 3. SSH hardening (روی کنترل نود)
sudo tee /etc/ssh/sshd_config.d/99-hardening.conf <<'EOF'
PermitRootLogin no
PasswordAuthentication no
KbdInteractiveAuthentication no
PubkeyAuthentication yes
AuthenticationMethods publickey
AllowUsers ansible
MaxAuthTries 3
LoginGraceTime 30
ClientAliveInterval 300
ClientAliveCountMax 2
X11Forwarding no
AllowAgentForwarding no
AllowTcpForwarding no
PermitTunnel no
EOF
sudo systemctl reload sshd

# 4. فایروال (فقط SSH از bastion)
sudo firewall-cmd --permanent --add-rich-rule='rule family="ipv4" source address="10.0.0.0/24" service name="ssh" accept'
sudo firewall-cmd --permanent --remove-service=ssh
sudo firewall-cmd --reload

# 5. auditd روی فایل‌های حساس
sudo tee /etc/audit/rules.d/ansible.rules <<'EOF'
-w /etc/ansible -p wa -k ansible_config
-w /opt/ansible-venv -p wa -k ansible_bin
-w /etc/sudoers.d -p wa -k sudoers_change
-w /home/ansible/.ssh -p wa -k ssh_keys
-w /var/log/ansible.log -p wa -k ansible_log
EOF
sudo augenrules --load

# 6. disable unused services
for svc in telnet.socket rsh.socket rlogin.socket tftp.socket; do
  sudo systemctl disable --now $svc 2>/dev/null || true
done

Separation of duties
نقش	دسترسی
Ansible developer	git push به branch، بدون اجرا روی prod
Ansible operator	اجرای playbook روی staging/prod (با approval)
Ansible admin	تغییر ansible.cfg، vault، نقش‌ها
Security team	audit log، review PR

مثال RBAC با AAP/AWX:

    Organization: ops

    Teams: developers (write staging)، operators (run prod)، admins (all)

    Credentials: جدا برای هر environment

    Inventories: read-only برای developers

اولویت‌ها (بالا به پایین — بالاتر برنده)
text

1. ANSIBLE_* env vars       ⚠️ خطر: مهاجم می‌تواند set کند
2. CLI args                 ⚠️ تاریخچه shell
3. $ANSIBLE_CONFIG
4. ./ansible.cfg            ⚠️ اگر cwd attacker-controlled
5. ~/.ansible.cfg
6. /etc/ansible/ansible.cfg ✅ فقط در production

توصیه: در production، فقط /etc/ansible/ansible.cfg استفاده کن. ANSIBLE_* env vars را در shell profiles بلاک کن:
bash

# /etc/profile.d/ansible-lock.sh
unset ANSIBLE_CONFIG ANSIBLE_INVENTORY ANSIBLE_VAULT_PASSWORD_FILE
unset ANSIBLE_HOST_KEY_CHECKING ANSIBLE_PRIVATE_KEY_FILE

✅ الگوی درست: SSH Key + CA
bash

# 1. تولید کلید Ed25519 (نه RSA)
ssh-keygen -t ed25519 -a 100 -C "ansible-deploy@control" -f ~/.ssh/ansible_deploy
chmod 600 ~/.ssh/ansible_deploy

# 2. SSH CA برای host verification (مقیاس‌پذیر)
# روی CA server
ssh-keygen -t ed25519 -f /root/ssh_ca -C "ssh-host-ca"

# امضای host key هر سرور
ssh-keygen -s /root/ssh_ca -I "web01" -h -n "web01,web01.example.com,10.0.0.1" \
  -V +52w /etc/ssh/ssh_host_ed25519_key.pub
# → /etc/ssh/ssh_host_ed25519_key-cert.pub

# 3. توزیع CA public key
sudo tee /etc/ssh/ssh_known_hosts.d/ca <<'EOF'
@cert-authority *.example.com ssh-ed25519 AAAA...
EOF

# 4. client config
sudo tee /etc/ssh/ssh_config.d/ansible.conf <<'EOF'
Host *.example.com
  HostKeyAlias %h
  StrictHostKeyChecking yes
  UserKnownHostsFile /etc/ssh/ssh_known_hosts
  PreferredAuthentications publickey
  PasswordAuthentication no
  KbdInteractiveAuthentication no
  IdentitiesOnly yes
  IdentityFile /etc/ansible/keys/ansible_deploy
EOF

SSH Agent + ProxyJump (bastion)
bash

# از bastion برای دسترسی به internal
cat > ~/.ssh/config <<'EOF'
Host bastion.example.com
  HostName bastion.example.com
  User jump
  IdentityFile /etc/ansible/keys/ansible_deploy
  IdentitiesOnly yes

Host *.internal
  ProxyJump bastion.example.com
  User deploy
  IdentityFile /etc/ansible/keys/ansible_deploy
  StrictHostKeyChecking yes
EOF

yaml

# inventory
all:
  vars:
    ansible_ssh_common_args: '-o ProxyJump=bastion.example.com -o StrictHostKeyChecking=yes'

- name: deploy sshd hardening

    src: sshd_config.j2

- name: test ssh before restart

ansible ALL=(root) NOPASSWD: /usr/bin/systemctl restart nginx, /usr/bin/systemctl reload nginx
ansible ALL=(root) NOPASSWD: /usr/bin/dnf, /usr/bin/rpm

# با alias برای خوانایی
Cmnd_Alias PKG = /usr/bin/apt, /usr/bin/apt-get, /usr/bin/dpkg, /usr/bin/dnf, /usr/bin/rpm
Cmnd_Alias SVC = /usr/bin/systemctl restart nginx, /usr/bin/systemctl reload nginx
ansible ALL=(root) NOPASSWD: PKG, SVC

Encrypt با vault-id (multi-environment)
bash

# رمز production vault در HashiCorp Vault
cat > /etc/ansible/vault-prod.sh <<'EOF'
#!/bin/bash
vault kv get -field=password secret/ansible/vault-prod
EOF
chmod 700 /etc/ansible/vault-prod.sh

# encrypt با id مشخص
ansible-vault encrypt --vault-id prod@/etc/ansible/vault-prod.sh \
  group_vars/prod/vault.yml

# فایل شروع می‌شود با:
# $ANSIBLE_VAULT;1.2;AES256;prod

# اجرا
ansible-playbook site.yml --vault-id prod@/etc/ansible/vault-prod.sh

Ansible callback برای audit
python

# /etc/ansible/callback_plugins/audit_logger.py
from ansible.plugins.callback import CallbackBase
import json, logging, os

class CallbackModule(CallbackBase):
    CALLBACK_VERSION = 2.0
    CALLBACK_TYPE = 'notification'
    CALLBACK_NAME = 'audit_logger'
    CALLBACK_NEEDS_WHITELIST = True

    def __init__(self):
        super().__init__()
        self.logger = logging.getLogger('ansible_audit')
        handler = logging.FileHandler('/var/log/ansible-audit.json')
        handler.setFormatter(logging.Formatter('%(message)s'))
        self.logger.addHandler(handler)
        self.logger.setLevel(logging.INFO)

    def _log(self, event, result):
        entry = {
            'event': event,
            'task': result._task.get_name() if result._task else None,
            'host': result._host.get_name() if result._host else None,
            'result': {
                'changed': getattr(result, '_result', {}).get('changed'),
                'failed': getattr(result, '_result', {}).get('failed'),
            },
            'user': os.environ.get('USER', 'unknown'),
            'playbook': self._playbook,
        }
        self.logger.info(json.dumps(entry))

    def v2_playbook_on_start(self, playbook):
        self._playbook = playbook._file_name

    def v2_runner_on_ok(self, result):
        self._log('task_ok', result)

    def v2_runner_on_failed(self, result, ignore_errors=False):
        self._log('task_failed', result)

    def v2_runner_on_unreachable(self, result):
        self._log('unreachable', result)

# retry با until
- name: wait for service
  uri:
    url: "http://localhost:8080/health"
  register: h
  until: h.status == 200
  retries: 30
  delay: 3
  failed_when: h.status not in [200, 503]  # 503 = در حال راه‌اندازی

# block/rescue برای rollback
- block:
    - name: deploy
      copy: src=app dest=/opt/app/app
      notify: restart app
    - meta: flush_handlers
    - name: health
      uri: url=http://localhost/health
      retries: 10
      delay: 3
  rescue:
    - name: restore previous version
      copy: src=app.old dest=/opt/app/app
    - systemd: name=app state=restarted
    - fail: msg="Deployment failed, rolled back"

# any_errors_fatal در cluster
- hosts: cluster
  any_errors_fatal: true
  max_fail_percentage: 0    # یک fail = abort

# meta end_play برای abort تمیز
- meta: end_play
  when: preflight_failed

# callback_plugins/prometheus.py
from ansible.plugins.callback import CallbackBase
from prometheus_client import Counter, Histogram
import time

TASK_COUNT = Counter('ansible_tasks_total', 'Tasks', ['status'])
TASK_DURATION = Histogram('ansible_task_duration_seconds', 'Duration', ['task'])

class CallbackModule(CallbackBase):
    CALLBACK_NAME = 'prometheus'
    CALLBACK_NEEDS_WHITELIST = True

    def v2_runner_on_ok(self, result):
        TASK_COUNT.labels(status='ok').inc()

    def v2_runner_on_failed(self, result, ignore_errors=False):
        TASK_COUNT.labels(status='failed').inc()

    def v2_runner_on_unreachable(self, result):
        TASK_COUNT.labels(status='unreachable').inc()
