1️⃣ ansible — Ad-hoc Commands
کارکرد

اجرای یک ماژول روی یک یا چند هاست بدون نوشتن playbook. مناسب برای کارهای سریع یک‌بارمصرف.
سینتکس
bash

ansible <pattern> [-m MODULE] [-a "ARGS"] [options]

گزینه‌های کلیدی
گزینه	توضیح
-m, --module-name	ماژول (پیش‌فرض: command)
-a, --args	آرگومان‌های ماژول به‌صورت key=value یا JSON
-i, --inventory	مسیر inventory
-u, --user	کاربر SSH
-b, --become	فعال‌سازی privilege escalation
-K, --ask-become-pass	پرسیدن رمز sudo
-k, --ask-pass	پرسیدن رمز SSH
-l, --limit	محدود کردن به زیرمجموعه‌ای
-f, --forks	تعداد موازی (پیش‌فرض 5)
-e, --extra-vars	متغیرهای اضافی
-T, --timeout	timeout اتصال
-B, --background	async timeout
-P, --poll	poll interval (پیش‌فرض 15)
-C, --check	dry-run
-D, --diff	نمایش تفاوت
-v/-vv/-vvv/-vvvv	سطح verbose
--private-key	کلید SSH
--list-hosts	فقط لیست هاست‌ها
--tree	ذخیره خروجی در دایرکتوری
--syntax-check	بررسی سینتکس
--step	تأیید هر task
--start-at-task	شروع از یک task
--ssh-common-args	آرگومان‌های SSH
--ssh-extra-args	
-o, --one-line	خروجی تک‌خطی
-t, --tree	
مثال‌های عملی
پینگ همه هاست‌ها
bash

ansible all -m ping
ansible all -i inventory -m ping -u deploy --private-key ~/.ssh/id_rsa

بررسی uptime
bash

ansible webservers -m command -a "uptime"
ansible all -a "uptime" -o   # خروجی یک‌خطی

نصب پکیج با privilege escalation
bash

ansible all -b -m apt -a "name=htop state=present"
ansible all -b -K -m yum -a "name=httpd state=latest"

کپی فایل
bash

ansible all -m copy -a "src=/tmp/config dest=/etc/app/config mode=0644 owner=root"

اجرای shell command
bash

ansible all -m shell -a "ps aux | grep nginx"
ansible all -m command -a "/usr/bin/systemctl status nginx"

مدیریت سرویس
bash

ansible all -b -m service -a "name=nginx state=restarted"
ansible all -b -m systemd -a "name=nginx state=started enabled=yes"

Fact gathering
bash

ansible host -m setup
ansible host -m setup -a "filter=ansible_*ipv4*"
ansible host -m setup -a "gather_subset=network"

با محدودیت
bash

ansible all -l web01 -m ping
ansible webservers -l "web01:web02" -m ping

با متغیر
bash

ansible all -e "target=/opt/myapp" -m file -a "path={{ target }} state=directory"

async (کارهای طولانی)
bash

ansible all -B 3600 -P 0 -m command -a "/usr/local/bin/backup.sh"
ansible all -B 3600 -P 0 -m command -a "long-task" -o

با SSH tunnel / bastion
bash

ansible all --ssh-common-args='-o ProxyJump=bastion.example.com' -m ping

بررسی اتصال با verbose
bash

ansible all -m ping -vvv

دانلود فایل
bash

ansible all -m fetch -a "src=/var/log/syslog dest=/tmp/logs/ flat=no"

اجرای اسکریپت
bash

ansible all -m script -a "/tmp/setup.sh arg1"

2️⃣ ansible-playbook — اجرای Playbook
کارکرد

اجرای فایل YAML حاوی play/task/role.
سینتکس
bash

ansible-playbook <playbook.yml> [options]

گزینه‌های کلیدی
گزینه	توضیح
-i, --inventory	inventory
-l, --limit	محدود کردن هاست‌ها
-e, --extra-vars	متغیرها
--extra-vars @file.yml	متغیرها از فایل
--vault-password-file	رمز vault
--ask-vault-pass	پرسیدن رمز
--vault-id	id:source
-C, --check	dry-run
-D, --diff	نمایش تفاوت
--syntax-check	بررسی سینتکس
--list-tasks	لیست taskها
--list-tags	لیست tagها
--list-hosts	لیست هاست‌ها
--start-at-task	شروع از task مشخص
--step	تأیید هر task
-t, --tags	اجرای tagهای مشخص
--skip-tags	نادیده گرفتن tagها
-f, --forks	موازی
--flush-cache	پاک کردن fact cache
--force-handlers	اجرای handlerها حتی در fail
--force-color	رنگ اجباری
-b, --become	privilege escalation
-K	پرسیدن رمز sudo
--timeout	SSH timeout
--check + --diff	ترکیب برای audit
--private-key	کلید
--limit	
--inventory-file	منسوخ
-v	verbose (تا ۴ بار)
--notify	(نادر)
--list-tasks	
--connection	نوع اتصال
--module-path	مسیر ماژول‌های اضافی
--start-at-task	
--force-color	
--list-tags	
--skip-tags	
مثال‌های عملی
اجرای ساده
bash

ansible-playbook site.yml
ansible-playbook -i inventory/prod site.yml

با متغیر
bash

ansible-playbook site.yml -e "version=1.2.3"
ansible-playbook site.yml -e @vars/prod.yml
ansible-playbook site.yml -e '{"version": "1.2.3", "env": "prod"}'

Dry-run + diff
bash

ansible-playbook site.yml --check --diff

بررسی سینتکس
bash

ansible-playbook site.yml --syntax-check

لیست
bash

ansible-playbook site.yml --list-tasks
ansible-playbook site.yml --list-hosts
ansible-playbook site.yml --list-tags

اجرای tag
bash

ansible-playbook site.yml --tags "deploy,restart"
ansible-playbook site.yml --tags "install"
ansible-playbook site.yml --skip-tags "test,debug"

شروع از task
bash

ansible-playbook site.yml --start-at-task "Deploy artifact"

تأیید مرحله‌ای (بی‌نهایت مفید در prod)
bash

ansible-playbook site.yml --step

محدود به زیرمجموعه
bash

ansible-playbook site.yml --limit "web01:web02"
ansible-playbook site.yml --limit "webservers:&prod"
ansible-playbook site.yml --limit "@retry_hosts.txt"

با vault
bash

ansible-playbook site.yml --ask-vault-pass
ansible-playbook site.yml --vault-password-file=~/.vault_pass
ansible-playbook site.yml --vault-id prod@vault-client.py
ansible-playbook site.yml --vault-id dev@prompt --vault-id prod@vault-prod.sh

با become
bash

ansible-playbook site.yml -b -K

با verbose
bash

ansible-playbook site.yml -vvv
ansible-playbook site.yml -vvvv --start-at-task "failing task"

با callback
bash

ANSIBLE_STDOUT_CALLBACK=json ansible-playbook site.yml
ANSIBLE_STDOUT_CALLBACK=yaml ansible-playbook site.yml

retry
bash

ansible-playbook site.yml
# پس از شکست، فایل .retry ساخته می‌شود
ansible-playbook site.yml --limit @site.retry

دیباگ حفظ فایل‌های remote
bash

ANSIBLE_KEEP_REMOTE_FILES=1 ansible-playbook site.yml -vvv

اجرای playbook از URL
bash

ansible-playbook https://example.com/playbook.yml

با fork محدود
bash

ansible-playbook site.yml -f 10

9️⃣ ansible-pull — Pull Mode
کارکرد

هاست‌ها خودشان playbook را از git می‌کشند و روی خودشان اجرا می‌کنند (برعکس push mode).
مثال‌ها
pull پایه
bash

ansible-pull -U https://github.com/org/repo.git playbook.yml

با checkout branch
bash

ansible-pull -U https://github.com/org/repo.git -C production site.yml

با inventory و limit
bash

ansible-pull -U https://github.com/org/repo.git \
  -i localhost, \
  -l localhost \
  -d /var/lib/ansible/local \
  site.yml

بدون git (local)
bash

ansible-pull -U file:///srv/playbooks playbook.yml

زمان‌بندی با cron
bash

# /etc/cron.d/ansible-pull
*/15 * * * * root /usr/bin/ansible-pull \
  -U https://github.com/org/repo.git \
  -C production \
  -i /etc/ansible/hosts \
  -o \
  /var/lib/ansible/local \
  site.yml >> /var/log/ansible-pull.log 2>&1

گزینه‌های مهم:

    -U: URL مخزن

    -C: branch

    -d: دایرکتوری clone

    -o: --only-if-changed (فقط اگر repo تغییر کرده)

    -i: inventory

    -s: sleep بین اجرا

    --accept-host-key

1️⃣4️⃣ ansible-runner — Python API/CLI
کارکرد

اجرای Ansible به‌صورت برنامه‌نویسی‌شده (Python library + CLI).
مثال‌ها
CLI
bash

ansible-runner run /path/to/project -p playbook.yml
ansible-runner run /path/to/project -p site.yml -i inventory
ansible-runner run . -p playbook.yml --cmdline "-vvv"

Python API
python

import ansible_runner

r = ansible_runner.run(
    private_data_dir='/path/to/project',
    playbook='site.yml',
    inventory='inventory',
    extravars={'env': 'prod'},
    limit='webservers',
)
print(f"{r.status}: {r.rc}")
for event in r.events:
    print(event.get('event'))

با callbacks
python

def my_handler(event):
    print(event.get('event'), event.get('task'))

r = ansible_runner.run(
    private_data_dir='.',
    playbook='site.yml',
    event_handler=my_handler,
)

چرخه کامل یک deploy
bash

# 1. نصب وابستگی‌ها
ansible-galaxy collection install -r requirements.yml
ansible-galaxy role install -r roles/requirements.yml -p roles/

# 2. رمزگشایی
ansible-vault view --vault-password-file ~/.vault_pass group_vars/prod/vault.yml

# 3. بررسی
ansible-playbook site.yml --syntax-check
ansible-playbook site.yml --check --diff --limit staging

# 4. اجرا
ansible-playbook site.yml -e @vars/prod.yml --vault-password-file ~/.vault_pass

# 5. اجرا روی failed فقط
ansible-playbook site.yml --limit @site.retry

# 6. بررسی نتیجه
ansible all -i inventory/prod -m shell -a "systemctl is-active nginx"

چرخه یک Incident
bash

# دیباگ اتصال

# فکت‌ها

# لاگ

# کانفیگ
ansible-config dump --only-changed

# inventory
ansible-inventory -i inventory --host web01

چرخه CI/CD
bash

# 1. lint

# 2. syntax

# 3. check

# 4. deploy
ansible-playbook -i inventory/prod site.yml \
  --vault-password-file=vault-client.py \
  -e "env=prod version=${CI_COMMIT_SHA}"

# 5. verify
