# streaming.py
import ansible_runner

def event_handler(event):
    """callback برای هر event"""
    event_type = event.get('event')
    
    if event_type == 'playbook_on_start':
        print(f"▶️  Playbook: {event.get('playbook')}")
    
    elif event_type == 'playbook_on_play_start':
        print(f"\n🎬 Play: {event.get('play', {}).get('name')}")
    
    elif event_type == 'playbook_on_task_start':
        print(f"  ▶ {event.get('task', {}).get('name')}")
    
    elif event_type == 'runner_on_ok':
        host = event.get('event_data', {}).get('host')
        result = event.get('event_data', {}).get('res', {})
        status = "changed" if result.get('changed') else "ok"
        print(f"    ✅ [{host}] {status}")
    
    elif event_type == 'runner_on_failed':
        host = event.get('event_data', {}).get('host')
        msg = event.get('event_data', {}).get('res', {}).get('msg', '')
        print(f"    ❌ [{host}] {msg}")
    
    elif event_type == 'runner_on_unreachable':
        host = event.get('event_data', {}).get('host')
        print(f"    🔌 [{host}] UNREACHABLE")
    
    elif event_type == 'playbook_on_stats':
        stats = event.get('event_data', {})
        print(f"\n📊 Stats: {stats}")

r = ansible_runner.run(
    private_data_dir='/opt/ansible/project',
    playbook='site.yml',
    event_handler=event_handler,
)

if r.rc != 0:
    raise SystemExit(f"Playbook failed with rc={r.rc}")

# read_artifact.py
import json
from pathlib import Path

def parse_artifact(artifact_dir: str) -> dict:
    """parse output کامل یک اجرا"""
    base = Path(artifact_dir)
    
    result = {
        'status': (base / 'status').read_text().strip(),
        'rc': int((base / 'rc').read_text().strip()),
        'stats': {},
        'tasks': [],
        'failures': [],
    }
    
    # Parse events
    for event_file in (base / 'job_events').glob('*.json'):
        event = json.loads(event_file.read_text())
        event_type = event.get('event')
        
        if event_type == 'playbook_on_stats':
            result['stats'] = event['event_data']
        
        elif event_type == 'runner_on_ok':
            data = event.get('event_data', {})
            result['tasks'].append({
                'host': data.get('host'),
                'task': data.get('task'),
                'changed': data.get('res', {}).get('changed', False),
            })
        
        elif event_type == 'runner_on_failed':
            data = event.get('event_data', {})
            result['failures'].append({
                'host': data.get('host'),
                'task': data.get('task'),
                'msg': data.get('res', {}).get('msg', ''),
            })
    
    return result

# quick_command.py
import ansible_runner

def run_command(hosts: str, cmd: str, become: bool = False) -> dict:
    """اجرای shell command روی hosts"""
    r = ansible_runner.run(
        private_data_dir='/tmp/runner',
        host_pattern=hosts,
        module='command',
        module_args=cmd,
        inventory='/etc/ansible/inventory',
        envvars={
            'ANSIBLE_HOST_KEY_CHECKING': 'True',
            'ANSIBLE_STDOUT_CALLBACK': 'json',
        },
        # become
        cmdline='-b' if become else '',
    )
    
    results = {}
    for event in r.events:
        if event.get('event') == 'runner_on_ok':
            data = event['event_data']
            results[data['host']] = {
                'rc': data['res'].get('rc', 0),
                'stdout': data['res'].get('stdout', ''),
                'stderr': data['res'].get('stderr', ''),
            }
        elif event.get('event') == 'runner_on_failed':
            data = event['event_data']
            results[data['host']] = {
                'error': data['res'].get('msg', 'unknown'),
            }
    
    return results

# parallel_runs.py
from concurrent.futures import ThreadPoolExecutor, as_completed
import ansible_runner
import os

def deploy_host(host: str, version: str) -> dict:
    """deploy روی یک هاست"""
    r = ansible_runner.run(
        private_data_dir='/opt/ansible/project',
        playbook='deploy.yml',
        inventory='/etc/ansible/inventory',
        limit=host,
        extravars={'app_version': version, 'env': 'prod'},
        quiet=True,
        timeout=1800,
    )
    return {
        'host': host,
        'status': r.status,
        'rc': r.rc,
        'artifact': r.artifact_dir,
    }

def deploy_all(hosts: list, version: str, concurrency: int = 5) -> list:
    """deploy با concurrency محدود"""
    results = []
    
    with ThreadPoolExecutor(max_workers=concurrency) as executor:
        futures = {
            executor.submit(deploy_host, host, version): host
            for host in hosts
        }
        
        for future in as_completed(futures):
            host = futures[future]
            try:
                result = future.result(timeout=1900)
                results.append(result)
                
                icon = "✅" if result['status'] == 'successful' else "❌"
                print(f"{icon} {host}: {result['status']}")
            except Exception as e:
                print(f"❌ {host}: exception {e}")
                results.append({'host': host, 'status': 'error', 'error': str(e)})
    
    return results

# استفاده
hosts = ['web01', 'web02', 'web03', 'web04', 'web05']
results = deploy_all(hosts, version='1.2.3', concurrency=3)

# گزارش
failed = [r for r in results if r['status'] != 'successful']
if failed:
    print(f"\n❌ Failed hosts: {[r['host'] for r in failed]}")
    exit(1)

# async_deploy.py
import asyncio
import ansible_runner
from concurrent.futures import ThreadPoolExecutor

# ansible-runner sync است → از thread pool استفاده کن
executor = ThreadPoolExecutor(max_workers=10)

async def async_deploy(host: str, version: str) -> dict:
    loop = asyncio.get_event_loop()
    return await loop.run_in_executor(
        executor,
        lambda: ansible_runner.run(
            private_data_dir='/opt/ansible/project',
            playbook='deploy.yml',
            limit=host,
            extravars={'app_version': version},
            quiet=True,
        )
    )

async def deploy_all_async(hosts: list, version: str) -> list:
    tasks = [async_deploy(h, version) for h in hosts]
    return await asyncio.gather(*tasks, return_exceptions=True)

# استفاده

    results = await deploy_all_async(hosts, '1.2.3')
    for host, result in zip(hosts, results):
        if isinstance(result, Exception):
            print(f"❌ {host}: {result}")
        else:
            print(f"✅ {host}: {result.status}")

#!/usr/bin/python
# library/my_custom_module.py
from ansible.module_utils.basic import AnsibleModule

DOCUMENTATION = r'''
---
module: my_custom_module
short_description: Manage custom app
description:
  - Idempotent custom app management
options:
  name:
    description: App name
    required: true
    type: str
  version:
    description: Version
    required: false
    type: str
    default: "latest"
  state:
    description: desired state
    choices: [present, absent, latest]
    default: present
    type: str
'''

EXAMPLES = r'''
- name: Install app
  my_custom_module:
    name: myapp
    version: "1.2.3"
    state: present
'''

RETURN = r'''
version:
  description: installed version
  type: str
  returned: always
'''

def get_installed_version(name: str) -> str:
    """خواندن نسخه نصب‌شده از filesystem"""
    import os
    path = f'/opt/{name}/.version'
    if not os.path.exists(path):
        return None
    with open(path) as f:
        return f.read().strip()

def install_app(name: str, version: str) -> tuple:
    """نصب app — return (changed, msg)"""
    import subprocess
    cmd = ['/usr/local/bin/install-app', '--name', name, '--version', version]
    result = subprocess.run(cmd, capture_output=True, text=True)
    return (result.returncode == 0, result.stdout + result.stderr)

def main():
    module = AnsibleModule(
        argument_spec=dict(
            name=dict(type='str', required=True),
            version=dict(type='str', default='latest'),
            state=dict(type='str', default='present', choices=['present', 'absent', 'latest']),
        ),
        supports_check_mode=True,   # ← مهم!
    )
    
    name = module.params['name']
    version = module.params['version']
    state = module.params['state']
    
    current = get_installed_version(name)
    changed = False
    msg = ''
    
    # Check mode (dry-run)
    if module.check_mode:
        if state == 'absent' and current:
            module.exit_json(changed=True, msg=f"Would remove {name}")
        elif state == 'present' and current != version:
            module.exit_json(changed=True, msg=f"Would install {name}-{version}")
        module.exit_json(changed=False, msg=f"{name} is {state}")
    
    # Real execution
    if state == 'absent':
        if current:
            import os, shutil
            shutil.rmtree(f'/opt/{name}', ignore_errors=True)
            changed = True
            msg = f"Removed {name}"
    
    elif state in ('present', 'latest'):
        if state == 'latest':
            version = 'latest'   # install-app handles it
        
        if current != version:
            success, output = install_app(name, version)
            if not success:
                module.fail_json(msg=f"Install failed: {output}")
            changed = True
            msg = f"Installed {name}-{version}"
    
    module.exit_json(
        changed=changed,
        msg=msg,
        version=get_installed_version(name),
    )

if __name__ == '__main__':
    main()

# filter_plugins/custom_filters.py
class FilterModule:
    def filters(self):
        return {
            'nginx_upstreams': self.nginx_upstreams,
            'b64_to_json': self.b64_to_json,
            'mask_secrets': self.mask_secrets,
            'docker_tag': self.docker_tag,
        }

    def nginx_upstreams(self, hosts: list) -> str:
        """تبدیل لیست هاست‌ها به upstream nginx"""
        lines = []
        for i, host in enumerate(hosts):
            lines.append(f'    server {host}:8080 max_fails=3 fail_timeout=10s;')
        return '\n'.join(lines)

    def b64_to_json(self, b64_str: str) -> dict:
        """decode base64 + parse JSON"""
        import base64, json
        return json.loads(base64.b64decode(b64_str))

    def mask_secrets(self, text: str) -> str:
        """ماسک کردن secret در لاگ"""
        import re
        patterns = [
            (r'(password|passwd|pwd)[=:]\s*[^\s,]+', r'\1=***'),
            (r'(token|key|secret)[=:]\s*[^\s,]+', r'\1=***'),
            (r'Bearer\s+[A-Za-z0-9\-._~+/]+=*', 'Bearer ***'),
        ]
        for pattern, replacement in patterns:
            text = re.sub(pattern, replacement, text, flags=re.IGNORECASE)
        return text

    def docker_tag(self, version: str, env: str) -> str:
        """compute docker tag"""
        # 1.2.3 + prod → 1.2.3-prod
        # 1.2.3 + staging → 1.2.3-rc
        suffix = {'prod': 'prod', 'staging': 'rc', 'dev': 'dev'}.get(env, 'latest')
        return f"{version}-{suffix}"

# lookup_plugins/aws_secret.py
from ansible.plugins.lookup import LookupBase
import boto3

class LookupModule(LookupBase):
    def run(self, terms, variables=None, **kwargs):
        """
        terms: لیست secret_id ها
        return: لیست value ها
        """
        region = kwargs.get('region', 'us-east-1')
        client = boto3.client('secretsmanager', region_name=region)
        
        results = []
        for secret_id in terms:
            try:
                response = client.get_secret_value(SecretId=secret_id)
                results.append(response['SecretString'])
            except client.exceptions.ResourceNotFoundException:
                raise Exception(f"Secret not found: {secret_id}")
        
        return results

        vault_token = kwargs.get('vault_token') or os.environ['VAULT_TOKEN']

            r = requests.get(url, headers=headers, timeout=10)
            r.raise_for_status()
            results.append(r.json()['data']['data'])
        
        return results

from ansible.plugins.callback import CallbackBase
from prometheus_client import Counter, Histogram, Gauge
import time
import os

TASK_COUNT = Counter(
    'ansible_task_total',
    'Total tasks executed',
    ['status', 'playbook', 'play']
)
TASK_DURATION = Histogram(
    'ansible_task_duration_seconds',
    'Task execution duration',
    ['task']
)
HOSTS_BY_STATUS = Gauge(
    'ansible_hosts_by_status',
    'Hosts by status',
    ['status']
)

class CallbackModule(CallbackBase):
    CALLBACK_VERSION = 2.0
    CALLBACK_TYPE = 'aggregate'
    CALLBACK_NAME = 'prometheus'
    CALLBACK_NEEDS_WHITELIST = True

    def __init__(self):
        super().__init__()
        self._playbook = None
        self._play = None
        self._task_start = {}

    def v2_playbook_on_start(self, playbook):
        self._playbook = os.path.basename(playbook._file_name)

    def v2_playbook_on_play_start(self, play):
        self._play = play.get_name()

    def v2_playbook_on_task_start(self, task, is_conditional):
        self._task_start[task._uuid] = time.time()

    def _duration(self, task):
        start = self._task_start.pop(task._uuid, None)
        if start:
            TASK_DURATION.labels(task=task.get_name()).observe(time.time() - start)

    def v2_runner_on_ok(self, result):
        self._duration(result._task)
        TASK_COUNT.labels(
            status='ok',
            playbook=self._playbook,
            play=self._play
        ).inc()

    def v2_runner_on_failed(self, result, ignore_errors=False):
        self._duration(result._task)
        status = 'ignored' if ignore_errors else 'failed'
        TASK_COUNT.labels(
            status=status,
            playbook=self._playbook,
            play=self._play
        ).inc()

    def v2_runner_on_unreachable(self, result):
        TASK_COUNT.labels(
            status='unreachable',
            playbook=self._playbook,
            play=self._play
        ).inc()

    def v2_runner_on_skipped(self, result):
        TASK_COUNT.labels(
            status='skipped',
            playbook=self._playbook,
            play=self._play
        ).inc()

#!/usr/bin/env python3
# inventory/dynamic_inventory.py
"""
Dynamic inventory from internal CMDB.
Usage: ./dynamic_inventory.py --list
       ./dynamic_inventory.py --host <hostname>
"""
import json
import sys
import argparse
import requests

CMDB_API = 'https://cmdb.internal/api/v1'

def get_servers():
    r = requests.get(f'{CMDB_API}/servers', timeout=10)
    r.raise_for_status()
    return r.json()

def build_inventory(servers: list) -> dict:
    inventory = {
        '_meta': {'hostvars': {}},
    }
    
    for srv in servers:
        host = srv['hostname']
        group = srv['role']           # webserver, db, cache
        env = srv['environment']      # prod, staging
        
        # گروه‌ها: role و env
        inventory.setdefault(group, {'hosts': []})
        inventory.setdefault(env, {'hosts': []})
        inventory[group]['hosts'].append(host)
        inventory[env]['hosts'].append(host)
        
        # hostvars
        inventory['_meta']['hostvars'][host] = {
            'ansible_host': srv['ip'],
            'ansible_user': 'deploy',
            'ansible_ssh_private_key_file': '/etc/ansible/keys/deploy',
            'environment': env,
            'role': group,
            'tags': srv.get('tags', {}),
        }
    
    # Composite groups
    for role in ['webserver', 'db', 'cache']:
        for env in ['prod', 'staging']:
            group_name = f'{role}_{env}'
            hosts = [
                h for h, v in inventory['_meta']['hostvars'].items()
                if v['role'] == role and v['environment'] == env
            ]
            if hosts:
                inventory[group_name] = {'hosts': hosts}
    
    return inventory

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--list', action='store_true')
    parser.add_argument('--host', help='Get host variables')
    args = parser.parse_args()
    
    if args.list:
        servers = get_servers()
        print(json.dumps(build_inventory(servers), indent=2))
    elif args.host:
        servers = get_servers()
        inventory = build_inventory(servers)
        hostvars = inventory['_meta']['hostvars'].get(args.host, {})
        print(json.dumps(hostvars, indent=2))
    else:
        parser.print_help()
        sys.exit(1)

if __name__ == '__main__':
    main()

#!/usr/bin/env python3
# daily/health_check.py
"""
Daily health check across all production servers.
Run: cron 0 8 * * * /opt/scripts/daily/health_check.py
"""
import ansible_runner
import json
import logging
import sys
from datetime import datetime
from pathlib import Path
import requests

logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s [%(levelname)s] %(message)s',
    handlers=[
        logging.FileHandler('/var/log/ansible/health_check.log'),
        logging.StreamHandler(),
    ]
)
logger = logging.getLogger(__name__)

PLAYBOOK = '/opt/ansible/project/project/health_check.yml'
INVENTORY = '/etc/ansible/inventory/production'
SLACK_WEBHOOK = 'https://hooks.slack.com/services/XXX'


def run_health_check() -> dict:
    logger.info("Starting daily health check")
    
    r = ansible_runner.run(
        private_data_dir='/opt/ansible/project',
        playbook='health_check.yml',
        inventory=INVENTORY,
        envvars={
            'ANSIBLE_HOST_KEY_CHECKING': 'True',
            'ANSIBLE_STDOUT_CALLBACK': 'json',
            'ANSIBLE_FORKS': '20',
        },
        timeout=900,
    )
    
    # Parse results
    results = {
        'timestamp': datetime.utcnow().isoformat(),
        'status': r.status,
        'rc': r.rc,
        'hosts': {},
        'issues': [],
    }
    
    for event in r.events:
        event_type = event.get('event')
        data = event.get('event_data', {})
        host = data.get('host')
        
        if event_type == 'runner_on_ok':
            if host not in results['hosts']:
                results['hosts'][host] = {'status': 'ok', 'checks': {}}
            
            task = data.get('task', '')
            res = data.get('res', {})
            results['hosts'][host]['checks'][task] = res
        
        elif event_type == 'runner_on_failed':
            if host not in results['hosts']:
                results['hosts'][host] = {'status': 'ok', 'checks': {}}
            results['hosts'][host]['status'] = 'failed'
            results['issues'].append({
                'host': host,
                'task': data.get('task'),
                'msg': data.get('res', {}).get('msg', ''),
            })
        
        elif event_type == 'runner_on_unreachable':
            if host not in results['hosts']:
                results['hosts'][host] = {'status': 'ok', 'checks': {}}
            results['hosts'][host]['status'] = 'unreachable'
            results['issues'].append({
                'host': host,
                'task': data.get('task'),
                'msg': 'Host unreachable',
            })
    
    return results


def analyze_results(results: dict) -> dict:
    """تحلیل نتایج و استخراج هشدارها"""
    alerts = {
        'critical': [],
        'warning': [],
        'info': [],
    }
    
    for host, info in results['hosts'].items():
        if info['status'] == 'unreachable':
            alerts['critical'].append(f"🔌 {host}: unreachable")
            continue
        
        if info['status'] == 'failed':
            alerts['critical'].append(f"❌ {host}: failed")
            continue
        
        # تحلیل disk
        disk = info['checks'].get('check disk usage', {})
        # فرض: task با assert برگشته
        if disk.get('failed'):
            alerts['warning'].append(f"💾 {host}: low disk")
        
        # تحلیل memory
        mem = info['checks'].get('check memory', {})
        if mem.get('failed'):
            alerts['warning'].append(f"🧠 {host}: low memory")
    
    return alerts


def send_slack_alert(alerts: dict, results: dict):
    """ارسال به Slack"""
    total = len(results['hosts'])
    ok = sum(1 for h in results['hosts'].values() if h['status'] == 'ok')
    
    emoji = "✅" if not alerts['critical'] else "🚨"
    
    blocks = [
        {
            "type": "header",
            "text": {"type": "plain_text", "text": f"{emoji} Daily Health Check"}
        },
        {
            "type": "section",
            "fields": [
                {"type": "mrkdwn", "text": f"*Total Hosts:* {total}"},
                {"type": "mrkdwn", "text": f"*OK:* {ok}"},
                {"type": "mrkdwn", "text": f"*Critical:* {len(alerts['critical'])}"},
                {"type": "mrkdwn", "text": f"*Warnings:* {len(alerts['warning'])}"},
            ]
        }
    ]
    
    if alerts['critical']:
        blocks.append({
            "type": "section",
            "text": {"type": "mrkdwn", "text": "*CRITICAL:*\n" + "\n".join(alerts['critical'][:10])}
        })
    
    if alerts['warning']:
        blocks.append({
            "type": "section",
            "text": {"type": "mrkdwn", "text": "*Warnings:*\n" + "\n".join(alerts['warning'][:10])}
        })
    
    try:
        requests.post(SLACK_WEBHOOK, json={"blocks": blocks}, timeout=10)
    except Exception as e:
        logger.error(f"Slack failed: {e}")


def main():
    try:
        results = run_health_check()
        logger.info(f"Health check: {results['status']}, hosts={len(results['hosts'])}")

        try:
            import psycopg2
            conn = psycopg2.connect(
                host='db.internal', database='monitoring',
                user='ansible', password=os.environ['DB_PASS']
            )
            cur = conn.cursor()
            cur.execute("""
                INSERT INTO health_checks (timestamp, status, total_hosts, ok_hosts, issues)
                VALUES (%s, %s, %s, %s, %s)
            """, (
                results['timestamp'],
                results['status'],
                len(results['hosts']),
                sum(1 for h in results['hosts'].values() if h['status'] == 'ok'),
                json.dumps(results['issues']),
            ))
            conn.commit()
            conn.close()
        except Exception as e:
            logger.error(f"DB write failed: {e}")
        
        alerts = analyze_results(results)
        
        # فقط اگر critical هست، به Slack
        if alerts['critical'] or alerts['warning']:
            send_slack_alert(alerts, results)

        if alerts['critical']:
            sys.exit(2)
        elif alerts['warning']:
            sys.exit(1)

        logger.exception(f"Health check crashed: {e}")
        send_slack_alert({'critical': [f"Health check crashed: {e}"], 'warning': [], 'info': []}, {'hosts': {}})
        sys.exit(3)


if __name__ == '__main__':
    import os

#!/usr/bin/env python3
# daily/deploy.py
"""
Multi-environment deployment with canary + health checks + rollback.
Usage: ./deploy.py --version 1.2.3 --env production
"""
import argparse
import ansible_runner
import json
import logging
import sys
import time
from datetime import datetime
from pathlib import Path
import requests
import psycopg2

logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s [%(levelname)s] %(message)s',
)
logger = logging.getLogger(__name__)

PRIVATE_DATA_DIR = '/opt/ansible/project'
INVENTORY = {
    'staging': '/etc/ansible/inventory/staging',
    'production': '/etc/ansible/inventory/production',
}
SLACK_WEBHOOK = 'https://hooks.slack.com/services/XXX'


class DeployError(Exception):
    pass


def notify_slack(message: str, color: str = 'good'):
    blocks = [
        {
            "type": "section",
            "text": {"type": "mrkdwn", "text": message}
        }
    ]
    try:
        requests.post(
            SLACK_WEBHOOK,
            json={"attachments": [{"color": color, "blocks": blocks}]},
            timeout=5,
        )
    except Exception as e:
        logger.error(f"Slack: {e}")


def run_playbook(
    playbook: str,
    env: str,
    extravars: dict,
    limit: str = 'all',
    timeout: int = 1800,
) -> dict:
    """اجرای یک playbook و برگرداندن نتیجه"""
    logger.info(f"Running {playbook} on {env}/{limit}")
    
    r = ansible_runner.run(
        private_data_dir=PRIVATE_DATA_DIR,
        playbook=playbook,
        inventory=INVENTORY[env],
        limit=limit,
        extravars=extravars,
        envvars={
            'ANSIBLE_HOST_KEY_CHECKING': 'True',
            'ANSIBLE_STDOUT_CALLBACK': 'json',
            'ANSIBLE_FORKS': '10' if 'canary' in playbook else '5',
        },
        quiet=True,
        timeout=timeout,
    )
    
    result = {
        'playbook': playbook,
        'status': r.status,
        'rc': r.rc,
        'stats': {},
        'failed_hosts': [],
        'artifact': r.artifact_dir,
    }
    
    for event in r.events:
        et = event.get('event')
        if et == 'playbook_on_stats':
            result['stats'] = event.get('event_data', {})
        elif et == 'runner_on_failed':
            data = event.get('event_data', {})
            result['failed_hosts'].append({
                'host': data.get('host'),
                'task': data.get('task'),
                'msg': data.get('res', {}).get('msg', ''),
            })
    
    return result


def get_canary_hosts(env: str, count: int = 1) -> list:
    """یک هاست به‌عنوان canary"""
    r = ansible_runner.run(
        private_data_dir=PRIVATE_DATA_DIR,
        host_pattern=f'webservers_{env}',
        module='ping',
        inventory=INVENTORY[env],
        quiet=True,
    )
    
    hosts = []
    for event in r.events:
        if event.get('event') == 'runner_on_ok':
            hosts.append(event['event_data']['host'])
    
    return hosts[:count]


def deploy_canary(version: str, env: str) -> dict:
    """Canary deployment"""
    canary_hosts = get_canary_hosts(env, count=1)
    if not canary_hosts:
        raise DeployError(f"No hosts found for {env}")
    
    logger.info(f"Canary: {canary_hosts}")
    
    result = run_playbook(
        playbook='deploy.yml',
        env=env,
        extravars={
            'app_version': version,
            'deploy_env': env,
            'is_canary': True,
        },
        limit=canary_hosts[0],
    )
    
    if result['status'] != 'successful':
        raise DeployError(f"Canary failed: {result['failed_hosts']}")
    
    return result


def validate_canary(version: str, env: str):
    """تست canary به مدت ۵ دقیقه"""
    logger.info("Validating canary for 5 minutes...")
    
    canary_hosts = get_canary_hosts(env, count=1)
    host = canary_hosts[0]
    
    # بررسی health در ۵ دقیقه
    for i in range(10):   # 10 × 30s = 5 min
        r = ansible_runner.run(
            private_data_dir=PRIVATE_DATA_DIR,
            host_pattern=host,
            module='uri',
            module_args={
                'url': f'http://{host}/health',
                'status_code': 200,
                'timeout': 5,
            },
            inventory=INVENTORY[env],
            quiet=True,
        )
        
        if r.rc != 0:
            raise DeployError(f"Health check failed at iteration {i}")
        
        # بررسی error rate (اگر Prometheus داری)
        try:
            resp = requests.get(
                f'http://prometheus.internal/api/v1/query',
                params={'query': f'rate(http_requests_total{{host="{host}",status=~"5.."}}[1m])'},
                timeout=5,
            )
            value = float(resp.json()['data']['result'][0]['value'][1])
            if value > 0.01:  # بیش از 1% خطا
                raise DeployError(f"Error rate high on canary: {value}")
        except (IndexError, KeyError):
            pass  # no data
        
        time.sleep(30)
    
    logger.info("Canary validated ✅")


def deploy_rolling(version: str, env: str, batch_size: int = 5) -> dict:
    """Rolling deployment بعد از canary"""
    logger.info(f"Rolling deploy: batches of {batch_size}")
    
    result = run_playbook(
        playbook='deploy.yml',
        env=env,
        extravars={
            'app_version': version,
            'deploy_env': env,
            'is_canary': False,
            'serial': str(batch_size),
            'max_fail_percentage': 10,
        },
        limit='webservers',
        timeout=3600,
    )
    
    return result


def record_deployment(version: str, env: str, result: dict, started: str):
    """ذخیره در DB"""
    try:
        import os
        conn = psycopg2.connect(
            host='db.internal', database='deployments',
            user='deploy', password=os.environ['DB_PASS'],
        )
        cur = conn.cursor()
        cur.execute("""
            INSERT INTO deployments 
            (version, env, status, started_at, finished_at, stats, failed_hosts, artifact)
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s)
        """, (
            version, env, result['status'],
            started, datetime.utcnow().isoformat(),
            json.dumps(result['stats']),
            json.dumps(result['failed_hosts']),
            result['artifact'],
        ))
        conn.commit()
        conn.close()
    except Exception as e:
        logger.error(f"DB record failed: {e}")


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--version', required=True)
    parser.add_argument('--env', required=True, choices=['staging', 'production'])
    parser.add_argument('--skip-canary', action='store_true')
    parser.add_argument('--batch-size', type=int, default=5)
    args = parser.parse_args()
    
    started = datetime.utcnow().isoformat()
    deploy_id = f"{args.env}-{args.version}-{datetime.now():%Y%m%d-%H%M%S}"
    
    logger.info(f"=== Deploy {deploy_id} ===")
    notify_slack(f"🚀 *Deploy started:* `{args.version}` → `{args.env}`")
    
    try:
        # 1. Canary
        if not args.skip_canary:
            deploy_canary(args.version, args.env)
            notify_slack(f"✅ Canary OK `{args.version}` on `{args.env}`")
            
            # 2. Validate
            validate_canary(args.version, args.env)
            notify_slack(f"✅ Canary validated after 5min `{args.version}`")
        
        # 3. Rolling
        result = deploy_rolling(args.version, args.env, args.batch_size)
        
        if result['status'] != 'successful':
            raise DeployError(f"Rolling failed: {result['failed_hosts']}")
        
        # 4. Record
        record_deployment(args.version, args.env, result, started)
        
        notify_slack(
            f"✅ *Deploy SUCCESS:* `{args.version}` → `{args.env}`\n"
            f"Stats: {result['stats']}"
        )

def get_current_version(env: str) -> str:
    """آخرین نسخه deploy شده"""
    conn = psycopg2.connect(

    cur.execute("""

        ORDER BY finished_at DESC

    conn.close()
    return row[0] if row else None


def get_previous_version(env: str, current: str) -> str:
    """نسخه قبل از current"""
    conn = psycopg2.connect(
        host='db.internal', database='deployments',
        user='deploy', password=os.environ['DB_PASS'],
    )
    cur = conn.cursor()
    cur.execute("""
        SELECT version FROM deployments
        WHERE env = %s AND status = 'successful' AND version != %s
        ORDER BY finished_at DESC
        LIMIT 1
    """, (env, current))
    row = cur.fetchone()
    conn.close()
    return row[0] if row else None


def rollback(env: str, target_version: str) -> dict:
    r = ansible_runner.run(
        private_data_dir='/opt/ansible/project',
        playbook='rollback.yml',
        inventory=f'/etc/ansible/inventory/{env}',

        },

            'ANSIBLE_STDOUT_CALLBACK': 'json',
        },

        timeout=1800,
    )

def extract_severity(task_name: str) -> str:
    """استخراج severity از نام task"""
    name = task_name.lower()
    if 'critical' in name or 'root' in name or 'firewall' in name:
        return 'high'
    elif 'warning' in name or 'config' in name:
        return 'medium'
    return 'low'

        'by_severity': {'high': 0, 'medium': 0, 'low': 0},
        'by_host': {},
        'top_findings': [],

    for result in all_findings:
        for finding in result['findings']:
            report['by_severity'][finding['severity']] += 1
            
            host = finding['host']
            report['by_host'].setdefault(host, []).append(finding)
    
    # Top 10 hosts با بیشترین finding
    report['top_hosts'] = sorted(
        report['by_host'].items(),
        key=lambda x: len(x[1]),
        reverse=True,
    )[:10]
    
    return report

#!/usr/bin/env python3
# weekly/capacity_report.py
"""
Weekly capacity analysis: CPU, memory, disk trends.
Predicts when resources will run out.
"""
import ansible_runner
import json
import logging
from datetime import datetime, timedelta
from pathlib import Path
from statistics import mean
import psycopg2

logger = logging.getLogger(__name__)


def collect_metrics(env: str) -> dict:
    """جمع‌آوری metric از همه هاست‌ها"""
    r = ansible_runner.run(
        private_data_dir='/opt/ansible/project',
        playbook='monitoring/collect-capacity.yml',
        inventory=f'/etc/ansible/inventory/{env}',
        envvars={
            'ANSIBLE_HOST_KEY_CHECKING': 'True',
            'ANSIBLE_STDOUT_CALLBACK': 'json',
            'ANSIBLE_FORKS': '20',
        },
        quiet=True,
        timeout=1800,
    )
    
    metrics = {}
    for event in r.events:
        if event.get('event') != 'runner_on_ok':
            continue
        
        data = event.get('event_data', {})
        host = data.get('host')
        task = data.get('task', '')
        res = data.get('res', {})
        
        if 'collect metrics' in task.lower():
            # فرض: module با stdout JSON برمی‌گرداند
            try:
                metrics[host] = json.loads(res.get('stdout', '{}'))
            except json.JSONDecodeError:
                logger.warning(f"{host}: invalid JSON")
    
    return metrics


def analyze_trends(env: str) -> dict:
    """تحلیل روند از داده‌های ۴ هفته گذشته"""
    conn = psycopg2.connect(
        host='db.internal', database='monitoring',
        user='ansible', password='xxx',
    )
    cur = conn.cursor()
    
    # query برای ۴ هفته اخیر
    cur.execute("""
        SELECT host, metric, date, value
        FROM capacity_metrics
        WHERE env = %s AND date > %s
        ORDER BY host, metric, date
    """, (env, (datetime.now() - timedelta(days=28)).date()))
    
    data = {}
    for host, metric, date, value in cur.fetchall():
        data.setdefault(host, {}).setdefault(metric, []).append((date, value))
    
    conn.close()
    
    # محاسبه رشد
    analysis = {}
    for host, metrics in data.items():
        analysis[host] = {}
        for metric, points in metrics.items():
            if len(points) < 2:
                continue
            
            values = [p[1] for p in points]
            dates = [p[0] for p in points]
            
            # linear regression ساده
            n = len(values)
            x = list(range(n))
            slope = (n * sum(i*v for i, v in zip(x, values)) - sum(x)*sum(values)) / (n*sum(i*i for i in x) - sum(x)**2) if n > 1 else 0
            
            current = values[-1]
            # روزهای باقی‌مانده تا 100% (اگر slope مثبت باشد)
            days_to_full = None
            if slope > 0 and metric.endswith('_pct'):
                days_to_full = int((100 - current) / slope) if current < 100 else 0
            
            analysis[host][metric] = {
                'current': current,
                'mean_4w': mean(values),
                'slope_per_day': round(slope, 3),
                'days_to_full': days_to_full,
            }
    
    return analysis


def generate_capacity_report(env: str) -> dict:
    """گزارش کامل capacity"""
    current = collect_metrics(env)
    trends = analyze_trends(env)
    
    report = {
        'env': env,
        'generated': datetime.utcnow().isoformat(),
        'current': current,
        'trends': trends,
        'warnings': [],
        'recommendations': [],
    }
    
    for host, metrics in trends.items():
        for metric, data in metrics.items():
            if data.get('days_to_full') and data['days_to_full'] < 30:
                report['warnings'].append({
                    'host': host,
                    'metric': metric,
                    'current_pct': data['current'],
                    'days_to_full': data['days_to_full'],
                    'severity': 'high' if data['days_to_full'] < 7 else 'medium',
                })
    
    # توصیه‌های scaling
    for host, metrics in trends.items():
        cpu_days = metrics.get('cpu_pct', {}).get('days_to_full')
        mem_days = metrics.get('memory_pct', {}).get('days_to_full')
        
        if cpu_days and cpu_days < 14:
            report['recommendations'].append(
                f"{host}: Add CPU cores within {cpu_days} days"
            )
        if mem_days and mem_days < 14:
            report['recommendations'].append(
                f"{host}: Add RAM within {mem_days} days"
            )
    
    return report

        requests.post(
            'https://hooks.slack.com/services/XXX',
            json={'text': msg},
            timeout=10,
        )

#!/usr/bin/env python3
# weekly/cost_optimization.py
"""
Weekly cost analysis for AWS resources.
Identifies idle/oversized resources.
"""
import boto3
import ansible_runner
import json
import logging
from datetime import datetime, timedelta

logger = logging.getLogger(__name__)


class CostAnalyzer:
    def __init__(self, region: str = 'us-east-1'):
        self.ec2 = boto3.client('ec2', region_name=region)
        self.cw = boto3.client('cloudwatch', region_name=region)
        self.pricing = boto3.client('pricing', region_name='us-east-1')
        self.region = region

    def find_idle_instances(self, days: int = 7) -> list:
        """EC2 instances با CPU < 5% در N روز گذشته"""
        idle = []
        
        # همه instances running
        instances = self.ec2.describe_instances(
            Filters=[
                {'Name': 'instance-state-name', 'Values': ['running']},
                {'Name': 'tag:Environment', 'Values': ['production', 'staging']},
            ]
        )
        
        for reservation in instances['Reservations']:
            for inst in reservation['Instances']:
                inst_id = inst['InstanceId']
                
                # میانگین CPU ۷ روز
                response = self.cw.get_metric_statistics(
                    Namespace='AWS/EC2',
                    MetricName='CPUUtilization',
                    Dimensions=[{'Name': 'InstanceId', 'Value': inst_id}],
                    StartTime=datetime.utcnow() - timedelta(days=days),
                    EndTime=datetime.utcnow(),
                    Period=3600,
                    Statistics=['Average'],
                )
                
                if response['Datapoints']:
                    avg_cpu = sum(d['Average'] for d in response['Datapoints']) / len(response['Datapoints'])
                    if avg_cpu < 5.0:
                        idle.append({
                            'instance_id': inst_id,
                            'type': inst['InstanceType'],
                            'avg_cpu': round(avg_cpu, 2),
                            'tags': {t['Key']: t['Value'] for t in inst.get('Tags', [])},
                        })
        
        return idle

    def estimate_monthly_savings(self, instances: list) -> float:
        """تخمین صرفه‌جویی ماهانه"""
        savings = 0
        for inst in instances:
            # قیمت تقریبی (باید از pricing API بگیرد)
            hourly_price = {
                't3.micro': 0.0104,
                't3.small': 0.0208,
                't3.medium': 0.0416,
                'm5.large': 0.096,
                'm5.xlarge': 0.192,
            }.get(inst['type'], 0.05)
            
            savings += hourly_price * 730   # ساعت در ماه
        
        return round(savings, 2)

    def find_oversized(self) -> list:
        """instances با CPU < 20% (کاندید downgrade)"""
        oversized = []
        
        instances = self.ec2.describe_instances(
            Filters=[{'Name': 'instance-state-name', 'Values': ['running']}]
        )
        
        for reservation in instances['Reservations']:
            for inst in reservation['Instances']:
                inst_id = inst['InstanceId']
                
                # Max CPU ۳۰ روز
                response = self.cw.get_metric_statistics(
                    Namespace='AWS/EC2',
                    MetricName='CPUUtilization',
                    Dimensions=[{'Name': 'InstanceId', 'Value': inst_id}],
                    StartTime=datetime.utcnow() - timedelta(days=30),
                    EndTime=datetime.utcnow(),
                    Period=3600,
                    Statistics=['Maximum', 'Average'],
                )
                
                if response['Datapoints']:
                    max_cpu = max(d['Maximum'] for d in response['Datapoints'])
                    avg_cpu = sum(d['Average'] for d in response['Datapoints']) / len(response['Datapoints'])
                    
                    if max_cpu < 40 and avg_cpu < 15:
                        oversized.append({
                            'instance_id': inst_id,
                            'type': inst['InstanceType'],
                            'avg_cpu': round(avg_cpu, 2),
                            'max_cpu': round(max_cpu, 2),
                            'current_type': inst['InstanceType'],
                            'suggested_type': self._suggest_downgrade(inst['InstanceType']),
                        })
        
        return oversized

    def _suggest_downgrade(self, instance_type: str) -> str:
        downgrades = {
            'm5.xlarge': 'm5.large',
            'm5.large': 't3.medium',
            't3.large': 't3.medium',
            't3.medium': 't3.small',
            'c5.xlarge': 'c5.large',
        }
        return downgrades.get(instance_type, instance_type)

    def find_unattached_volumes(self) -> list:
        """EBS volumes بدون instance"""
        volumes = self.ec2.describe_volumes(
            Filters=[{'Name': 'status', 'Values': ['available']}]
        )
        
        return [
            {
                'volume_id': v['VolumeId'],
                'size': v['Size'],
                'type': v['VolumeType'],
                'created': v['CreateTime'].isoformat(),
            }
            for v in volumes['Volumes']
        ]


def generate_report(analyzer: CostAnalyzer) -> dict:
    idle = analyzer.find_idle_instances(days=7)
    oversized = analyzer.find_oversized()
    volumes = analyzer.find_unattached_volumes()
    
    total_savings = analyzer.estimate_monthly_savings(idle)
    
    report = {
        'generated': datetime.utcnow().isoformat(),
        'region': analyzer.region,
        'idle_instances': idle,
        'oversized': oversized,
        'unattached_volumes': volumes,
        'estimated_monthly_savings': total_savings,
    }
    
    return report


def save_report(report: dict):
    from pathlib import Path
    Path('/var/reports/cost').mkdir(parents=True, exist_ok=True)
    file = f"/var/reports/cost/cost-{report['generated'][:10]}.json"
    with open(file, 'w') as f:
        json.dump(report, f, indent=2)
    return file


def notify_finops(report: dict):
    import requests
    msg = f"""
💰 *Weekly Cost Report*

*Idle instances:* {len(report['idle_instances'])} (${report['estimated_monthly_savings']}/mo potential)
*Oversized:* {len(report['oversized'])}
*Unattached volumes:* {len(report['unattached_volumes'])}

Top idle:
""" + '\n'.join(f"• `{i['instance_id']}` ({i['type']}) — CPU {i['avg_cpu']}%"

    requests.post(
        'https://hooks.slack.com/services/FINOPS',
        json={'text': msg},
        timeout=10,
    )


def main():
    logging.basicConfig(level=logging.INFO)
    
    analyzer = CostAnalyzer(region='us-east-1')

if __name__ == '__main__':
    main()

def execute_job(job_id: str, playbook: str, inventory: str,
                extravars: dict, limit: str, dry_run: bool):
    """Background task"""
    try:
        cmdline = '--check --diff' if dry_run else ''
        
        r = ansible_runner.run(
            private_data_dir='/opt/ansible/project',
            playbook=playbook,
            inventory=inventory,
            limit=limit,
            extravars=extravars,
            cmdline=cmdline,
            envvars={
                'ANSIBLE_HOST_KEY_CHECKING': 'True',
                'ANSIBLE_STDOUT_CALLBACK': 'json',
                'ANSIBLE_FORKS': '10',
            },
            quiet=True,
            timeout=3600,
        )
        
        JOBS[job_id]['status'] = r.status
        JOBS[job_id]['rc'] = r.rc
        JOBS[job_id]['finished_at'] = datetime.utcnow().isoformat()
        JOBS[job_id]['artifact'] = r.artifact_dir
        
        # Parse stats
        stats = {}
        failures = []
        for event in r.events:
            if event.get('event') == 'playbook_on_stats':
                stats = event.get('event_data', {})
            elif event.get('event') == 'runner_on_failed':
                data = event.get('event_data', {})
                failures.append({
                    'host': data.get('host'),
                    'task': data.get('task'),
                })
        
        JOBS[job_id]['stats'] = stats
        JOBS[job_id]['failures'] = failures
    
    except Exception as e:
        logger.exception("Job failed")
        JOBS[job_id]['status'] = 'error'
        JOBS[job_id]['error'] = str(e)


@app.post("/deploy", response_model=JobResponse, status_code=202)

    user: str = Depends(verify_token),
):

        'limit': req.limit,

        'user': user,
        'status': 'running',

        'app_version': req.version,
        'deploy_env': req.env,
    }
    
    background.add_task(
        execute_job, job_id, 'deploy.yml', inventory,
        extravars, req.limit, req.dry_run,
    )
    
    return JobResponse(
        job_id=job_id,
        status='running',
        submitted_at=JOBS[job_id]['submitted_at'],
    )

# tasks/ansible_tasks.py
from celery import Celery
import ansible_runner
import json

app = Celery('ansible_tasks', broker='redis://redis:6379/0', backend='redis://redis:6379/1')
app.conf.task_routes = {'deploy.*': {'queue': 'deploy'}}
app.conf.task_time_limit = 3600
app.conf.task_soft_time_limit = 3300


@app.task(bind=True, max_retries=3)
def deploy_task(self, host: str, version: str, env: str):
    """deploy یک هاست"""
    try:
        r = ansible_runner.run(
            private_data_dir='/opt/ansible/project',
            playbook='deploy.yml',
            inventory=f'/etc/ansible/inventory/{env}',
            limit=host,
            extravars={'app_version': version},
            envvars={'ANSIBLE_STDOUT_CALLBACK': 'json'},
            quiet=True,
        )
        
        if r.status == 'successful':
            return {'host': host, 'status': 'ok', 'rc': 0}
        
        # retry با exponential backoff
        raise self.retry(
            exc=Exception(f"Deploy failed: {r.status}"),
            countdown=2 ** self.request.retries * 30,
        )
    
    except Exception as e:
        return {'host': host, 'status': 'failed', 'error': str(e)}


@app.task
def deploy_all_task(version: str, env: str, hosts: list):
    """deploy موازی روی همه هاست‌ها"""
    from celery import group
    
    job = group(deploy_task.s(host, version, env) for host in hosts)
    result = job.apply_async()
    results = result.get(timeout=3600)
    
    failed = [r for r in results if r['status'] != 'ok']
    return {
        'total': len(hosts),
        'succeeded': len(hosts) - len(failed),
        'failed': failed,
    }

WEBHOOK_SECRET = os.environ['GITHUB_WEBHOOK_SECRET']

def verify_signature(payload: bytes, signature: str) -> bool:
    expected = 'sha256=' + hmac.new(
        WEBHOOK_SECRET.encode(),
        payload,
        hashlib.sha256,
    ).hexdigest()
    return hmac.compare_digest(expected, signature)


@app.post("/webhook/github")
async def github_webhook(request: Request):
    body = await request.body()
    signature = request.headers.get('X-Hub-Signature-256', '')
    
    if not verify_signature(body, signature):
        raise HTTPException(401, "Invalid signature")
    
    payload = await request.json()
    event = request.headers.get('X-GitHub-Event')
    
    # فقط push به tag
    if event == 'push' and payload.get('ref', '').startswith('refs/tags/v'):

            version=version,
            env='staging',
            hosts=['web01', 'web02', 'web03'],
        )
        
        return {'status': 'queued', 'version': version}
    
    return {'status': 'ignored'}

from prometheus_client import start_http_server, Gauge, Counter
import ansible_runner
import time
import logging

logger = logging.getLogger(__name__)

HOST_OK = Gauge('ansible_host_ok', 'Host is OK (1/0)', ['host', 'env'])
HOST_LAST_RUN = Gauge('ansible_host_last_run_timestamp', 'Last run', ['host', 'env'])
DEPLOY_COUNT = Counter('ansible_deploy_total', 'Deploys', ['status', 'env'])


def check_health(env: str, inventory: str):
    r = ansible_runner.run(
        private_data_dir='/opt/ansible/project',
        host_pattern='all',
        module='ping',
        inventory=inventory,
        envvars={'ANSIBLE_STDOUT_CALLBACK': 'json'},
        quiet=True,
    )
    
    for event in r.events:
        et = event.get('event')
        data = event.get('event_data', {})
        host = data.get('host')
        
        if not host:
            continue
        
        if et == 'runner_on_ok':
            HOST_OK.labels(host=host, env=env).set(1)
            HOST_LAST_RUN.labels(host=host, env=env).set(time.time())
        elif et in ('runner_on_failed', 'runner_on_unreachable'):
            HOST_OK.labels(host=host, env=env).set(0)
            HOST_LAST_RUN.labels(host=host, env=env).set(time.time())


def main():
    start_http_server(9100)
    
    while True:
        try:
            check_health('production', '/etc/ansible/inventory/production')
            check_health('staging', '/etc/ansible/inventory/staging')
        except Exception as e:
            logger.error(f"Export failed: {e}")
        
        time.sleep(300)


if __name__ == '__main__':
    logging.basicConfig(level=logging.INFO)
    main()

def read_secret(path: str) -> str:
    import os, stat
    st = os.stat(path)
    if st.st_mode & stat.S_IROTH:
        raise PermissionError(f"{path} is world-readable!")
    with open(path) as f:
        return f.read().strip()

import logging
import re

class SecretFilter(logging.Filter):
    PATTERNS = [
        (re.compile(r'(password|passwd|pwd)[=:]\s*\S+', re.I), r'\1=***'),
        (re.compile(r'(token|api[_-]?key)[=:]\s*\S+', re.I), r'\1=***'),
        (re.compile(r'Bearer\s+[\w\-._~+/]+=*'), 'Bearer ***'),
        (re.compile(r'AKIA[0-9A-Z]{16}'), 'AKIA***'),  # AWS
    ]
    
    def filter(self, record):
        msg = record.getMessage()
        for pattern, replacement in self.PATTERNS:
            msg = pattern.sub(replacement, msg)
        record.msg = msg
        record.args = ()
        return True

logger = logging.getLogger()
logger.addFilter(SecretFilter())

۲۲.۷ Rate limiting

from functools import wraps
import time

def rate_limit(max_calls: int, period: int):
    calls = []
    def decorator(fn):
        @wraps(fn)
        def wrapper(*args, **kwargs):
            now = time.time()
            calls[:] = [c for c in calls if c > now - period]
            if len(calls) >= max_calls:
                sleep_for = period - (now - calls[0])
                time.sleep(sleep_for)
            calls.append(time.time())
            return fn(*args, **kwargs)
        return wrapper
    return decorator


@rate_limit(max_calls=10, period=60)   # 10 در دقیقه
def deploy(host: str):
    return ansible_runner.run(...)

def preflight_checks(playbook: str, inventory: str) -> bool:
    """بررسی‌های اولیه قبل از اجرای واقعی"""
    # 1. syntax check
    r = ansible_runner.run(
        private_data_dir='.',
        playbook=playbook,
        cmdline='--syntax-check',
        quiet=True,
    )
    if r.rc != 0:
        raise ValueError(f"Syntax error in {playbook}")
    
    # 2. inventory exists
    import os
    if not os.path.exists(inventory):
        raise FileNotFoundError(f"Inventory not found: {inventory}")
    
    # 3. run --check
    r = ansible_runner.run(
        private_data_dir='.',
        playbook=playbook,
        inventory=inventory,
        cmdline='--check --diff',
        quiet=True,
    )
    if r.rc != 0:
        raise ValueError(f"Check mode failed: {r.rc}")
    
    return True
