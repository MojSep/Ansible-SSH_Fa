---
- name: play name
  hosts: webservers
  become: true
  become_user: root
  gather_facts: true
  gather_subset: [min, network]
  serial: "25%"
  max_fail_percentage: 10
  any_errors_fatal: false
  ignore_unreachable: false
  strategy: linear
  order: inventory
  throttle: 0
  run_once: false
  check_mode: false
  diff: false
  force_handlers: false
  no_log: false

  vars:
    http_port: 80

  vars_files:
    - vars/common.yml
    - vars/{{ env }}.yml

  vars_prompt:
    - name: deploy_version
      prompt: "Version?"
      private: false
      default: "latest"

  module_defaults:
    ansible.builtin.apt:
      update_cache: yes

  environment:
    HTTP_PROXY: "{{ proxy }}"

  pre_tasks:
    - name: preflight
      assert:
        that: [ "ansible_os_family == 'Debian'" ]

  roles:
    - common
    - role: nginx
      vars:
        nginx_port: 80
      tags: [nginx]

  tasks:
    - name: main task
      ansible.builtin.debug:
        msg: "hello"

  post_tasks:
    - name: notify
      community.general.slack:
        msg: "Done"

  handlers:
    - name: restart nginx
      systemd:
        name: nginx
        state: restarted

  tags: [deploy]

۷️⃣ کلیدواژه‌های Task (کامل)
yaml

- name: descriptive name
  ansible.builtin.copy:
    src: file.conf
    dest: /etc/app/file.conf
    owner: root
    group: root
    mode: '0644'
    backup: yes
    validate: 'nginx -t -c %s'

  # کنترل اجرا
  become: true
  become_user: root
  become_method: sudo
  become_flags: '-H -S -n'

  # شرط و loop
  when: ansible_os_family == "Debian"
  loop: "{{ items }}"
  loop_control:
    label: "{{ item.name }}"
    index_var: idx
    pause: 2

  # register و retry
  register: result
  until: result is success
  retries: 3
  delay: 5

  # تغییر وضعیت
  changed_when: result.changed
  failed_when: result.rc != 0

  # اعلان
  notify:
    - restart nginx
    - reload systemd

  # tag
  tags: [config, nginx]

  # امنیت
  no_log: true

  # خطا
  ignore_errors: false
  ignore_unreachable: false

  # delegate
  delegate_to: localhost
  delegate_facts: false
  run_once: true

  # environment
  environment:
    HTTP_PROXY: "{{ proxy }}"

  # async
  async: 60
  poll: 5

  # throttle
  throttle: 1

  # connection
  connection: local

  # check
  check_mode: no
  diff: no

  # vars
  vars:
    local_var: value

۱۱️⃣ Loop و Condition
Loop
yaml

# ساده
- debug: msg="{{ item }}"
  loop: [1, 2, 3]

# List
- debug: msg="{{ item }}"
  loop: "{{ my_list }}"

# Dict
- debug: msg="{{ item.key }}={{ item.value }}"
  loop: "{{ my_dict | dict2items }}"

# Nested
- debug: msg="{{ item.0 }}-{{ item.1 }}"
  loop: "{{ list1 | product(list2) | list }}"

# Range
- debug: msg="{{ item }}"
  loop: "{{ range(1, 5) | list }}"

# with_index
- debug: msg="{{ idx }}: {{ item }}"
  loop: "{{ items }}"
  loop_control:
    index_var: idx

# with_label (نمایش فقط name)
- debug: msg="{{ item.name }}"
  loop: "{{ users }}"
  loop_control:
    label: "{{ item.name }}"

# pause بین iterations
- debug: msg="{{ item }}"
  loop: "{{ items }}"
  loop_control:
    pause: 2

# until success
- command: /usr/local/bin/try
  register: r
  until: r.rc == 0
  retries: 5
  delay: 3

# with complex
- template:
    src: "{{ item.src }}"
    dest: "{{ item.dest }}"
  loop:
    - { src: "a.j2", dest: "/etc/a" }
    - { src: "b.j2", dest: "/etc/b" }
  loop_control:
    label: "{{ item.dest }}"

# with index and dict
- user:
    name: "{{ item.key }}"
    uid: "{{ item.value }}"
  loop: "{{ dict2items(users) }}"

Condition
yaml

# ساده
- debug: msg="Debian"
  when: ansible_os_family == "Debian"

# چندگانه (AND)
- debug: msg="both"
  when:
    - ansible_os_family == "Debian"
    - ansible_distribution_major_version | int >= 22

# OR
- debug: msg="or"
  when: env == "prod" or env == "staging"

# NOT
- debug: msg="not"
  when: not skip_install

# in
- debug: msg="web"
  when: "'web' in group_names"

# variable defined
- debug: msg="defined"
  when: my_var is defined

# command result
- command: /usr/bin/test -f /etc/config
  register: check
  ignore_errors: yes
  changed_when: false

- debug: msg="exists"
  when: check.rc == 0

# complex
- debug: msg="complex"
  when:
    - (env == "prod" and version != "latest") or
      (env == "dev" and version == "latest")

when با changed
yaml

- template:
    src: nginx.j2
    dest: /etc/nginx/nginx.conf
  register: config

- systemd:
    name: nginx
    state: reloaded
  when: config.changed

۱۲️⃣ Block/Rescue/Always
yaml

- block:
    - name: install
      apt:
        name: nginx
        state: present

    - name: configure
      template:
        src: nginx.j2
        dest: /etc/nginx/nginx.conf
      notify: reload nginx

    - meta: flush_handlers

    - name: health check
      uri:
        url: http://localhost/health
        status_code: 200
      retries: 10
      delay: 3

  rescue:
    - name: rollback
      apt:
        name: nginx
        state: absent

    - name: notify
      slack:
        msg: "Deploy failed: {{ ansible_failed_result.msg }}"
      delegate_to: localhost

    - name: fail
      fail:
        msg: "Deployment aborted"

  always:
    - name: cleanup
      file:
        path: /tmp/deploy.lock
        state: absent

  become: true
  when: env == "production"

ansible_failed_result

rescue:
  - debug:
      msg: "Task {{ ansible_failed_task.name }} failed with {{ ansible_failed_result.msg }}"

۱۶️⃣ Delegation و Strategy
delegate_to
yaml

- name: task on another host
  command: /usr/bin/migration
  delegate_to: db-primary

- name: localhost task
  command: /usr/bin/script
  delegate_to: localhost
  connection: local
  run_once: true

- name: fact از هاست دیگر
  setup:
  delegate_to: "{{ other_host }}"
  delegate_facts: true

- name: notify cluster
  command: /usr/bin/notify
  run_once: true
  delegate_to: "{{ groups['cluster'][0] }}"

serial
yaml

- hosts: cluster
  serial: 1                     # one at a time
  serial: 5                     # batch of 5
  serial: "25%"                 # 25% of hosts
  serial: [1, 5, 10, "100%"]    # canary → full
  max_fail_percentage: 10       # abort اگر بیش از 10% fail

strategy
yaml

- hosts: all
  strategy: linear       # default, batch-level
  strategy: free         # fast hosts proceed
  strategy: host_pinned  # one host at a time

throttle
yaml

- name: rate-limited API
  uri:
    url: "{{ api_url }}"
  throttle: 5            # حداکثر 5 concurrent برای این task

run_once
yaml

- name: database migration
  command: /app/migrate
  run_once: true
  delegate_to: "{{ groups['db_primary'][0] }}"

async + poll
yaml

# fire and forget
- name: long task
  command: /usr/local/bin/backup.sh
  async: 3600
  poll: 0
  register: job

# wait later
- name: wait
  async_status:
    jid: "{{ job.ansible_job_id }}"
  register: result
  until: result.finished
  retries: 60
  delay: 60

    mode: '0644'

    src: nginx.j2
    dest: /etc/nginx/nginx.conf
    mode: '0644'
    validate: 'nginx -t -c %s'
    backup: yes
    trim_blocks: yes
    lstrip_blocks: yes

# file
- file: { path: /etc/app, state: directory, mode: '0755' }
- file: { path: /etc/old, state: absent }
- file: { src: /opt/app-v1, dest: /opt/app, state: link }
- file: { path: /tmp/x, state: touch }

# lineinfile
- lineinfile:
    path: /etc/ssh/sshd_config
    regexp: '^PermitRootLogin'
    line: 'PermitRootLogin no'
    state: present
    backup: yes
    validate: 'sshd -t -f %s'

# blockinfile
- blockinfile:
    path: /etc/ssh/sshd_config
    marker: "# {mark} ANSIBLE MANAGED"
    block: |
      PermitRootLogin no
      PasswordAuthentication no
    validate: 'sshd -t -f %s'

# replace
- replace:
    path: /etc/app/config
    regexp: '^old=(.*)$'
    replace: 'new=\1'

# fetch
- fetch:
    src: /var/log/app.log
    dest: /backup/{{ inventory_hostname }}/
    flat: no
    fail_on_missing: no

# synchronize (rsync)
- synchronize:
    src: ./dist/
    dest: /var/www/
    delete: yes
    rsync_opts: ["--exclude=.git"]
    checksum: yes
  delegate_to: "{{ inventory_hostname }}"

# stat
- stat: { path: /etc/config }
  register: st
- debug: msg="exists {{ st.stat.exists }}"

# find
- find:
    paths: [/var/log]
    patterns: ["*.log"]
    age: 30d
  register: logs

# unarchive
- unarchive:
    src: /tmp/app.tar.gz
    dest: /opt/app
    remote_src: yes
    creates: /opt/app/app

# archive
- archive:
