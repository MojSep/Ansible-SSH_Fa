۱.۲ روش venv (توصیه‌شده برای Production)
چرا؟

نصب
bash

# 1. ساخت venv اختصاصی
sudo mkdir -p /opt/ansible-venv
sudo chown $(whoami):$(whoami) /opt/ansible-venv

python3 -m venv /opt/ansible-venv
source /opt/ansible-venv/bin/activate

# 2. ارتقاء pip/setuptools/wheel
pip install --upgrade pip setuptools wheel

# 3. نصب Ansible
pip install 'ansible-core==2.17.6'

# یا bundle کامل
pip install 'ansible==10.5.0'

ساخت wrapper برای دسترسی سراسری
bash

sudo tee /usr/local/bin/ansible <<'EOF'
#!/bin/bash
exec /opt/ansible-venv/bin/ansible "$@"
EOF
sudo chmod +x /usr/local/bin/ansible

# تکرار برای ansible-playbook, ansible-galaxy, ansible-vault, ansible-config, ansible-doc, ansible-inventory, ansible-console, ansible-connection
for cmd in ansible-playbook ansible-galaxy ansible-vault ansible-config ansible-doc ansible-inventory ansible-console ansible-connection; do
  sudo tee /usr/local/bin/$cmd >/dev/null <<EOF
#!/bin/bash
exec /opt/ansible-venv/bin/$cmd "\$@"
EOF
  sudo chmod +x /usr/local/bin/$cmd
done

podman run --rm \
  -v ~/.ssh:/root/.ssh:ro \

Wrapper برای استفاده روزمره
bash

cat > /usr/local/bin/ansible-ee <<'EOF'
#!/bin/bash
exec podman run --rm -it \
  -v ~/.ssh:/root/.ssh:ro \
  -v $(pwd):/work:z \
  -v ~/.ansible:/root/.ansible:z \
  -w /work \
  --env ANSIBLE_CONFIG=/work/ansible.cfg \
  quay.io/ansible/creator-ee:latest \
  "$@"
EOF
chmod +x /usr/local/bin/ansible-ee

# استفاده

۱.۵ روش GitHub Release (بدون pip/PyPI)

مزیت: از PyPI وابسته نیست؛ می‌توانید روی mirror آینه کنید.
bash

# دانلود tarball رسمی
curl -L -o ansible-core.tar.gz \
  https://github.com/ansible/ansible/releases/download/v2.17.6/ansible-core-2.17.6.tar.gz

# verification
curl -L -o ansible-core.tar.gz.asc \
  https://github.com/ansible/ansible/releases/download/v2.17.6/ansible-core-2.17.6.tar.gz.asc
gpg --verify ansible-core.tar.gz.asc ansible-core.tar.gz

# نصب از tarball
pip install ./ansible-core.tar.gz

برای محیط تحریمی: استفاده از GitHub proxy
bash

# githack / ghproxy
curl -L -o ansible-core.tar.gz \
  https://ghproxy.com/https://github.com/ansible/ansible/releases/download/v2.17.6/ansible-core-2.17.6.tar.gz

# یا پروکسی گیت
git config --global url."https://ghproxy.com/https://github.com".insteadOf "https://github.com"

⚠️ این پروکسی‌ها امنیت تضمین‌شده ندارند — همیشه GPG signature را verify کن قبل از نصب.

تنظیم pip برای استفاده از mirror
bash

# موقت
pip install --index-url https://mirror-pypi.runflare.com/simple --trusted-host mirror-pypi.runflare.com ansible-core

# دائمی (فقط کاربر)
mkdir -p ~/.config/pip
cat > ~/.config/pip/pip.conf <<EOF
[global]
index-url = https://mirror-pypi.runflare.com/simple
extra-index-url = https://pypi.org/simple
trusted-host = mirror-pypi.runflare.com

[install]
upgrade-strategy = only-if-needed
EOF

# یا سراسری
sudo mkdir -p /etc/pip
sudo tee /etc/pip.conf <<EOF
[global]
index-url = https://mirror-pypi.runflare.com/simple
trusted-host = mirror-pypi.runflare.com
EOF

۲.۱ روش pip download / pip wheel (توصیه‌شده)
مرحله ۱: روی ماشین با اینترنت (bundle builder)
bash

# مشخص کردن پلتفرم مقصد
# از `pip debug --verbose` روی ماشین مقصد استفاده کن یا حدس بزن:
# Linux x86_64, Python 3.10 → manylinux_2_17_x86_64, cp310

mkdir -p /tmp/ansible-bundle

# دانلود تمام wheelها + وابستگی‌ها
pip download ansible-core==2.17.6 \
  --dest /tmp/ansible-bundle \
  --only-binary=:all: \
  --platform manylinux_2_17_x86_64 \
  --platform manylinux2014_x86_64 \
  --python-version 310 \
  --implementation cp \
  --abi cp310

# برای ansible bundle کامل
pip download ansible==10.5.0 \
  --dest /tmp/ansible-bundle \
  --only-binary=:all: \
  --platform manylinux2014_x86_64 \
  --python-version 310 \
  --implementation cp \
  --abi cp310

مرحله ۲: ساخت checksum و بسته
bash

cd /tmp
sha256sum ansible-bundle/* > ansible-bundle.sha256

# ساخت آرشیو
tar czf ansible-bundle-$(date +%Y%m%d).tgz ansible-bundle ansible-bundle.sha256

# امضا (اختیاری ولی توصیه‌شده)
gpg --detach-sign -a ansible-bundle-$(date +%Y%m%d).tgz

مرحله ۳: انتقال
bash

# با USB، scp، یا هر روش آفلاین
scp ansible-bundle-*.tgz ansible-bundle-*.tgz.asc user@offline-server:/tmp/

مرحله ۴: نصب روی ماشین آفلاین
bash

cd /tmp
tar xzf ansible-bundle-*.tgz

# تأیید امضا
gpg --verify ansible-bundle-*.tgz.asc ansible-bundle-*.tgz

# تأیید checksum
cd ansible-bundle && sha256sum -c ../ansible-bundle.sha256

# نصب آفلاین
cd /tmp
python3 -m venv /opt/ansible-venv
source /opt/ansible-venv/bin/activate
pip install --no-index --find-links=/tmp/ansible-bundle ansible-core==2.17.6

⚠️ نکات مهم

    --only-binary=:all: جلوی دانلود source package (که نیاز به build دارد) را می‌گیرد

    --platform و --abi باید دقیقاً با ماشین مقصد یکی باشد

    اگر پکیجی wheel نداشت، pip download نسخه source می‌آورد که روی مقصد به python3-dev, gcc نیاز دارد

    برای Ansible معمولاً همه چیز wheel دارد (pure Python)

تشخیص پلتفرم مقصد
bash

# روی ماشین مقصد (اگر دسترسی داری)
python3 -c "import pip._internal.utils.compatibility_tags as t; print([str(x) for x in t.get_supported()][:5])"

# یا از یونیک نام
python3 -m pip debug --verbose | grep -E 'cp3|abi3|manylinux'
