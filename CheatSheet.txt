۵️⃣ ساختار Playbook
اسکلت
yaml

---
- name: play name
  hosts: webservers
  become: true
  gather_facts: true
  serial: "20%"
  max_fail_percentage: 10
  any_errors_fatal: false
  strategy: linear
  order: inventory
  vars:
    http_port: 80
  vars_files:
    - vars/common.yml
  vars_prompt:
    - name: deploy_version
      prompt: "Version?"
      private: false
  pre_tasks:
    - name: check
      assert: { that: [ "ansible_os_family == 'Debian'" ] }
  roles:
    - role: common
      tags: [common]
    - role: nginx
      vars: { nginx_port: 80 }
  tasks:
    - name: task
      module: args
  post_tasks:
    - name: notify
      debug: msg="done"
  handlers:
    - name: restart nginx
      systemd: { name: nginx, state: restarted }

Task کامل
yaml

- name: descriptive name
  ansible.builtin.copy:
    src: file.conf
    dest: /etc/app/file.conf
    owner: root
    group: root
    mode: '0644'
    backup: yes
    validate: 'nginx -t -c %s'
  become: true
  become_user: root
  when: ansible_os_family == "Debian"
  loop: "{{ items }}"
  loop_control:
    label: "{{ item.name }}"
    index_var: idx
  register: result
  until: result is success
  retries: 3
  delay: 5
  changed_when: result.changed
  failed_when: result.rc != 0
  notify: restart nginx
  tags: [config, nginx]
  no_log: false
  ignore_errors: false
  delegate_to: localhost
  run_once: true
  environment:
    HTTP_PROXY: "{{ proxy }}"
  async: 60
  poll: 5
  throttle: 1

۶️⃣ متغیرها و Precedence
منابع (از کم به زیاد)
text

1.  role defaults/main.yml
2.  inventory file/script group vars
3.  inventory group_vars/all
4.  playbook group_vars/all
5.  inventory group_vars/* (specific)
6.  playbook group_vars/*
7.  inventory host_vars/*
8.  playbook host_vars/*
9.  host facts / cached set_facts
10. play vars
11. play vars_prompt
12. play vars_files
13. role vars/main.yml
14. block vars
15. task vars
16. include_vars
17. set_facts / registered vars
18. role (and include_role) params
19. include params
20. extra vars (-e) ⚠️ ALWAYS WINS

انواع
yaml

# string, int, bool
my_str: "hello"
my_int: 42
my_bool: true

# list
my_list: [a, b, c]

# dict
my_dict:
  key1: val1
  key2: val2

# multiline
my_text: |
  line1
  line2

# folded
my_text: >
  line1 line2

# null
my_null: null

# reference
ref: "{{ my_dict.key1 }}"

# concatenate
path: "{{ base }}/{{ sub }}"

# default
safe: "{{ undefined_var | default('fallback') }}"

set_fact و register
yaml

- name: register output
  command: /bin/echo hello
  register: result

- debug: var=result.stdout

- set_fact:
    is_debian: "{{ ansible_os_family == 'Debian' }}"

- set_fact:
    combined: "{{ dict1 | combine(dict2, recursive=True) }}"

- set_fact:
    cacheable: yes    # ذخیره در fact cache

Scope
yaml

# host-scoped
- set_fact: my_var=value

# play-scoped (all hosts)
- set_fact:
    cluster_nodes: "{{ groups['all'] | map('extract', hostvars, 'ansible_host') | list }}"

# shared via hostvars
- debug:
    var: hostvars['web01']['ansible_default_ipv4']['address']

دسته‌های اصلی
text

Files & Content
copy
yaml

    src: local.conf
    dest: /etc/app/config
    mode: '0644'
    owner: root
    group: root
    backup: yes
    validate: 'nginx -t -c %s'
- copy:
    content: "{{ config_content }}"
    dest: /etc/app/config
- copy:
    src: /etc/app/
    dest: /opt/backup/
    remote_src: yes

template (Jinja2)
yaml

- template:
    src: nginx.conf.j2
    dest: /etc/nginx/nginx.conf
    mode: '0644'
    validate: 'nginx -t -c %s'
    backup: yes

file
yaml

- file: { path: /etc/app, state: directory, mode: '0755' }
- file: { path: /etc/app/old, state: absent }
- file: { src: /etc/app/a, dest: /etc/app/b, state: link, force: yes }
- file: { path: /tmp/x, state: touch, modification_time: preserve, access_time: preserve }

lineinfile
yaml

- lineinfile:
    path: /etc/ssh/sshd_config
    regexp: '^PermitRootLogin'
    line: 'PermitRootLogin no'
    state: present
    backup: yes
    validate: 'sshd -t -f %s'

blockinfile

- blockinfile:

      PermitRootLogin no
      PasswordAuthentication no

۱۹️⃣ Anti-patterns (پرهیز)
❌ shell برای همه چیز

- shell: apt install nginx -y

# ✅
- apt: name=nginx state=present

❌ command بدون idempotency
yaml

# ❌
- command: /usr/local/bin/setup.sh

# ✅
- command: /usr/local/bin/setup.sh
  args: { creates: /opt/app/.setup_done }

❌ changed_when: false روی taskهای تغییریافته
yaml

# ❌
- command: systemctl restart nginx
  changed_when: false     # دروغ

# ✅
- systemd: name=nginx state=restarted
  notify: healthcheck

❌ gather_facts با all همیشه
yaml

# ❌ در 500 host
- hosts: all
  gather_facts: yes

# ✅
- hosts: all
  gather_facts: yes
  gather_subset: [min, network]

❌ become: true سراسری
yaml

# ❌
- hosts: all
  become: true

# ✅
- hosts: all
  tasks:
    - name: task requiring root
      apt: name=nginx state=present
      become: true

# ✅

❌ رمز hardcode

# ✅

no_log: true

❌ with_* قدیمی
yaml

# ✅

- include_tasks: tasks.yml    # dynamic

bash

# Syntax check

# Dry-run
ansible-playbook x.yml --check --diff

# Only tags
ansible-playbook x.yml --tags deploy

# Skip tags
ansible-playbook x.yml --skip-tags test

# Limited hosts
ansible-playbook x.yml --limit web01

# Retry
ansible-playbook x.yml --limit @x.retry

# Extra vars
ansible-playbook x.yml -e "version=1.2.3"

# Vars file
ansible-playbook x.yml -e @vars/prod.yml

# Vault
ansible-playbook x.yml --ask-vault-pass

# Start at
ansible-playbook x.yml --start-at-task="Deploy"

# List
ansible-playbook x.yml --list-tasks --list-tags --list-hosts

🎓 جمع‌بندی: ۱۵ قاعده طلایی

    همیشه --syntax-check قبل از اجرا

    --check --diff در staging قبل از prod

    serial برای هر چیزی که downtime دارد

    max_fail_percentage برای abort خودکار

    no_log: true روی هر task با secret

    Vault برای همه رمزها + secret manager برای رمز vault

    host_key_checking = True هرگز تغییر نده

    pipelining = True + !requiretty برای سرعت و امنیت

    Idempotency test: دو بار اجرا، بار دوم changed=0

    FQCN همیشه (ansible.builtin.copy نه copy)

    Collections را pin کن (requirements.yml با version)

    ansible-lint + yamllint در CI

    Monitoring + log_path + SIEM

    Backup + rollback plan قبل از هر deploy

    Runbook برای هر playbook production
