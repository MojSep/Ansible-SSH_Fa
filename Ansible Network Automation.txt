۱️⃣ تفاوت Ansible برای Network vs Server
چالش‌های خاص Network Devices
ویژگی	Server (Linux)	Network Device
Python روی دستگاه	✅ دارد	❌ ندارد
اجرای ماژول روی دستگاه	✅ ممکن	❌ غیرممکن
اتصال	SSH + اجرای Python	SSH/Telnet/API محدود
Fact gathering	/proc, /sys	فقط CLI command یا NETCONF
Idempotency	ماژول تضمین می‌کند	باید خودمان control کنیم
Rollback	snapshot/filesystem	configure replace یا commit confirmed
Test	Molecule/Docker	سخت‌تر (GNS3, EVE-NG, containerlab)
Connection	ssh, local	network_cli, netconf, httpapi
Connection persistence	جلسه SSH	persistent connection لازم
چه اتفاقی می‌افتد؟
text

┌──────────────────────────┐
│  Control Node            │
│                          │
│  1. Ansible reads config │
│  2. Renders commands     │
│  3. Connects via SSH     │
│  4. Pushes CLI commands  │
│  5. Parses response      │
│  6. Evaluates changed?   │
└────────────┬─────────────┘
             │
             │ SSH / NETCONF / HTTPS
             ▼
┌──────────────────────────┐
│  Network Device          │
│  (switch, router, fw)    │
│  - IOS / Junos / EOS     │
│  - No Python             │
│  - CLI / API             │
└──────────────────────────┘

نکته کلیدی: ماژول‌ها روی کنترل نود اجرا می‌شوند و فقط command/XML/JSON به دستگاه می‌فرستند.
۲️⃣ معماری Connection
چرا connection: ssh مستقیم کار نمی‌کند؟

وقتی ansible_connection: ssh باشد، Ansible سعی می‌کند یک اسکریپت Python روی مقصد اجرا کند. Network device:

    Python ندارد

    shell ندارد

    فقط CLI یا XML یا REST می‌فهمد

Persistent Connection

برای network_cli و netconf، Ansible یک پروسه پشتیبان روی کنترل نود اجرا می‌کند که جلسه SSH را زنده نگه می‌دارد:
text

┌────────────────┐
│ ansible-playbook│
└────────┬───────┘
         │
         ▼
┌──────────────────────────────────┐
│  ansible-connection (daemon)     │
│  - Persistent session             │
│  - Local unix socket              │
│  - Reuses SSH channel             │
└────────┬─────────────────────────┘
         │ persistent SSH
         ▼
┌────────────────┐
│ Network Device │
└────────────────┘

مسیر socket
text

~/.ansible/pc/<uuid>

پاک‌سازی stuck socket
bash

ls ~/.ansible/pc/
pkill -f ansible-connection
rm -rf ~/.ansible/pc/*

تنظیم
ini

# ansible.cfg
[persistent_connection]
connection_timeout = 60
command_timeout = 60
connect_timeout = 60
control_path_dir = ~/.ansible/pc

۳️⃣ Connection Plugins
۳.۱ network_cli — SSH به CLI

کاربرد: اکثر تجهیزات (Cisco IOS/NX-OS, Arista EOS, Juniper Junos, Aruba, VyOS).
yaml

# inventory/hosts.yml
all:
  children:
    cisco_switches:
      hosts:
        sw01:
          ansible_host: 10.0.0.1
      vars:
        ansible_connection: network_cli
        ansible_network_os: cisco.ios.ios   # FQCN
        ansible_user: ansible
        ansible_password: "{{ vault_sw_password }}"
        ansible_become: yes
        ansible_become_method: enable       # برای IOS

تنظیمات مهم
پارامتر	توضیح
ansible_network_os	FQCN یا نام کوتاه (ios, junos, eos, nxos)
ansible_become_method	enable (Cisco) یا sudo
ansible_network_cli_ssh_type	libssh (سریع) یا paramiko (پیش‌فرض)
ansible_command_timeout	timeout per command (60s)
ansible_connect_timeout	timeout اتصال (30s)
libssh برای سرعت
bash

pip install ansible-pylibssh

yaml

ansible_network_cli_ssh_type: libssh

سرعت ۳-۵ برابر بیشتر از paramiko.
۳.۲ netconf — XML API

کاربرد: Juniper Junos، Cisco IOS-XR/NX-OS (با netconf)، Nokia SR-OS.
yaml

sw02:
  ansible_host: 10.0.0.2
  ansible_connection: netconf
  ansible_network_os: junipernetworks.junos.junos
  ansible_user: ansible
  ansible_password: "{{ vault_pw }}"
  ansible_netconf_port: 830

مزایا:

    Transactional (commit/rollback)

    ساختاریافته (XML)

    Confirm-commit (rollback خودکار)

    سریع‌تر از CLI

معایب:

    باید در دستگاه فعال باشد

    پیچیده‌تر

فعال‌سازی روی Junos
text

set system services netconf ssh
set system services netconf traceoptions file netconf-trace

مثال
yaml

- name: configure interface via netconf
  junipernetworks.junos.junos_config:
    lines:
      - "set interfaces ge-0/0/1 unit 0 family inet address 10.0.0.1/24"
    comment: "Ansible managed"
    confirm_commit: 5        # rollback اگر تا 5 دقیقه commit نشود

۷️⃣ Command Modules (اجرای show)
ios_command / eos_command / junos_command
yaml

- name: show commands
  cisco.ios.ios_command:
    commands:
      - show version
      - show ip interface brief
      - show ip bgp summary
  register: output

- debug:
    var: output.stdout_lines

# اجرای یک command با wait_for (تا وقتی output pattern را دارد)
- name: wait for BGP
  cisco.ios.ios_command:
    commands:
      - command: show bgp ipv4 unicast summary
        wait_for: "state = Established"
        match: contains
        retries: 30
        interval: 10

# با prompt (برای command تعاملی)
- name: write memory
  cisco.ios.ios_command:
    commands:
      - command: copy running-config startup-config
        prompt: 'Destination filename \[startup-config\]\?'
        answer: "\r"

parse (با ansible.utils)
yaml

- name: parse show output
  cisco.ios.ios_command:
    commands: [show ip interface brief]
  register: intf

- name: extract
  ansible.utils.cli_parse:
    text: "{{ intf.stdout[0] }}"
    parser:
      name: ansible.netcommon.native
    set_fact: interfaces
  # یا
  ansible.utils.cli_parse:
    text: "{{ intf.stdout[0] }}"
    parser:
      name: ansible.utils.ttp
      template_path: templates/interfaces.ttp
    set_fact: interfaces

- debug:
    msg: "{{ interfaces }}"

Genie Parser (Cisco)
yaml

- name: parse with genie
  ansible.utils.cli_parse:
    text: "{{ intf.stdout[0] }}"
    parser:
      name: cisco.ios.genie
    set_fact: parsed

# parsed.interfaces.<name>.ip_address
# parsed.interfaces.<name>.status

۸️⃣ Config Modules
ios_config (و معادل‌های دیگر)
yaml

ios_config:
  lines:                      # لیست command
  parents:                    # context (e.g., interface Gi0/1)
  match:                      # line | exact | none (پیش‌فرض: line)
  replace:                    # line | block (پیش‌فرض: line)
  before:                     # command قبل از تغییر
  after:                      # command بعد از تغییر
  backup:                     # yes → save running-config
  backup_options:
    filename:                 # مسیر backup
    dir_path:                 # پوشه backup
  save_when:                  # always | never | modified | changed
  diff_against:               # startup | intended | session
  diff_ignore_lines:          # regex برای ignore در diff
  src:                        # فایل config کامل (جایگزین lines)

مثال ۱: تغییر ساده
yaml

- name: configure NTP
  cisco.ios.ios_config:
    lines:
      - ntp server 10.0.0.1
      - ntp server 10.0.0.2
    save_when: modified

مثال ۲: تغییر داخل interface
yaml

- name: configure interface
  cisco.ios.ios_config:
    lines:
      - description Uplink to Core
      - switchport mode trunk
      - switchport trunk allowed vlan 10,20,30
      - no shutdown
    parents:
      - interface GigabitEthernet1/0/1
    save_when: changed

مثال ۳: Context تودرتو
yaml

- name: configure BGP neighbor
  cisco.ios.ios_config:
    lines:
      - remote-as 65002
      - description Peer with ISP-A
      - address-family ipv4 unicast
      - activate
      - soft-reconfiguration inbound
    parents:
      - router bgp 65001
      - neighbor 192.168.1.1
    save_when: changed

مثال ۴: Idempotency با match
yaml

# match: line  → هر خطی که در config باشد skip می‌شود
# match: exact → خط دقیقاً باید match شود
# match: none  → همیشه اضافه می‌کند (غیر idempotent)

- name: strict match
  cisco.ios.ios_config:
    lines:
      - ntp server 10.0.0.1
    match: exact

مثال ۵: Backup قبل از تغییر
yaml

- name: backup and configure
  cisco.ios.ios_config:
    lines:
      - aaa new-model
      - aaa authentication login default group tacacs+ local
    backup: yes
    backup_options:
      dir_path: /backups/network/{{ inventory_hostname }}
      filename: "config-{{ ansible_date_time.iso8601 }}.cfg"
    save_when: changed

مثال ۶: Diff
yaml

- name: diff against startup
  cisco.ios.ios_config:
    lines:
      - hostname new-name
    diff_against: startup
    diff_ignore_lines:
      - "^! Last configuration change"
      - "^ntp clock-period"
  register: diff

- debug:
    var: diff.diff

۲۲.۲ Config Drift Detection (روزانه)
yaml

- hosts: network
  gather_facts: no

  tasks:
    - name: current config
      cisco.ios.ios_command:
        commands: [show running-config]
      register: current
      when: "'ios' in ansible_network_os"

    - name: baseline
      set_fact:
        baseline: "{{ lookup('file', 'baselines/' + inventory_hostname + '.cfg') }}"
      when: current is succeeded

    - name: diff
      set_fact:
        drift: "{{ current.stdout[0] | difference(baseline.split('\n')) }}"
      when: current is succeeded

    - name: alert
      slack:
        msg: "Config drift on {{ inventory_hostname }}: {{ drift | length }} lines"
      when: drift | length > 0
      delegate_to: localhost

۲۲.۳ Rollback روی خطا
yaml

- hosts: switches
  gather_facts: no
  serial: 1
  any_errors_fatal: true

  tasks:
    - block:
        - name: backup
          cisco.ios.ios_config:
            backup: yes

        - name: change
          cisco.ios.ios_config:
            lines:
              - "{{ item }}"
          loop: "{{ changes }}"

        - name: verify
          cisco.ios.ios_command:
            commands: [show ip interface brief]
            wait_for: "connected"
            match: contains
          retries: 5

      rescue:
        - name: rollback
          cisco.ios.ios_command:
            commands:
              - command: "configure replace flash:backup.cfg force"
                prompt: "\[confirm\]"
                answer: "y"

        - name: notify
          slack:
            msg: "Rolled back {{ inventory_hostname }}"
          delegate_to: localhost
